# #997 Pipeline stages editor

A pipeline-stages editor: four rows that reorder by grip or menu, each collapsing to a summary on a narrow screen and opening a name, kind and stale-days form.

When to use: Settings screens that edit an ordered list of records (stages, tags, loss reasons, quick replies) with inline fields.

## Facets

- Kind: component / layout
- Style: default
- Motion: subtle
- Interaction: drag, press, keyboard
- Layout: single-column
- Color mode: both
- Density: regular
- Tech: css (cost: css)
- Reduced motion: handled; keyboard: operable
- Components: EditableRow, IconButton, NumberField, Select, SortableList, TextField
- Also searched as: reorderable list, settings list editor, stage editor

## Spec

- `<SortableList items={stages} getId={(s) => s.id} getLabel={(s) => s.name} onreorder={reorder} label="Pipeline stages">` owns the grip, the drag and the reorder menu
- `async function reorder({ from, to }: SortableListReorder)` applies the move, saves it and restores `before` when the save fails
- `<EditableRow label={stage.name}>` holds the `TextField`, `Select` and `NumberField` and collapses when its own width is narrow

## Source

```svelte
<script lang="ts">
  import { Trash2 } from '@lucide/svelte';
  import { EditableRow, IconButton, NumberField, Select, SortableList, TextField } from '@codeatlasdev/ui';
  import type { SortableListReorder } from '@codeatlasdev/ui';

  type Stage = { id: string; name: string; kind: string; staleDays: number };
  let stages = $state<Stage[]>([
    { id: 'new', name: 'New', kind: 'open', staleDays: 3 },
    { id: 'talk', name: 'Talking', kind: 'open', staleDays: 7 },
    { id: 'won', name: 'Won', kind: 'won', staleDays: 0 },
    { id: 'lost', name: 'Lost', kind: 'lost', staleDays: 0 },
  ]);
  const kinds = [
    { value: 'open', label: 'Open' },
    { value: 'won', label: 'Won' },
    { value: 'lost', label: 'Lost' },
  ];

  // Optimistic reorder: apply it, persist, and put the old order back when the save fails.
  async function reorder({ from, to }: SortableListReorder) {
    const before = stages;
    const next = [...stages];
    const [moved] = next.splice(from, 1);
    next.splice(to, 0, moved);
    stages = next;
    try {
      await save(next);
    } catch {
      stages = before;
    }
  }
  const save = (_: Stage[]) => Promise.resolve();
</script>

<div class="stages">
  <SortableList items={stages} getId={(s) => s.id} getLabel={(s) => s.name} onreorder={reorder} label="Pipeline stages">
    {#snippet item(stage)}
      <EditableRow label={stage.name}>
        {#snippet summary()}<strong>{stage.name}</strong>{/snippet}
        <TextField label="Name" bind:value={stage.name} />
        <Select label="Kind" options={kinds} bind:value={stage.kind} />
        <NumberField label="Stale after (days)" bind:value={stage.staleDays} min={0} />
        {#snippet actions()}
          <IconButton variant="text" aria-label="Delete {stage.name}" onclick={() => (stages = stages.filter((s) => s.id !== stage.id))}>
            <Trash2 size={16} />
          </IconButton>
        {/snippet}
      </EditableRow>
    {/snippet}
  </SortableList>
</div>

<style>
  .stages { width: 100%; }
</style>
```
