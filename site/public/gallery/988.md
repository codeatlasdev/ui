# #988 Hear it

A sound toggle with volume, a profile picker (system, glass, brutalism, terminal) and a board of buttons that each play one cue.

When to use: Let people audition a sound profile before turning interface sound on in an app.

## Facets

- Kind: component / feedback
- Style: default
- Motion: subtle
- Interaction: press, keyboard
- Layout: stack
- Color mode: both
- Density: regular
- Tech: css (cost: css)
- Reduced motion: handled; keyboard: operable
- Components: Button, SegmentedControl, SoundToggle
- Also searched as: sound profile, audio feedback, ui sounds, cue board

## Spec

- `<SoundToggle volume />` next to a `SegmentedControl` bound to the profile
- `$effect(() => sound.use(PROFILES[profile]))` pins the picked profile and `onDestroy` releases it
- A `role="group"` board of outlined small `Button` items, each with `sound={cue}` for one cue name
- The cues are `press`, `toggle-on`, `toggle-off`, `success`, `error`, `warning`, `open`, `close` and `tab`

## Source

```svelte
<script lang="ts">
  import { Button, SegmentedControl } from '@codeatlasdev/ui';
  import { SoundToggle, brutal, glass, sound, system, terminal } from '@codeatlasdev/ui/sound';
  import type { CueName } from '@codeatlasdev/ui';
  import { onDestroy } from 'svelte';

  const PROFILES = { system, glass, brutal, terminal };
  const CUES: CueName[] = ['press', 'toggle-on', 'toggle-off', 'success', 'error', 'warning', 'open', 'close', 'tab'];

  let profile = $state<keyof typeof PROFILES>('system');
  $effect(() => sound.use(PROFILES[profile]));
  onDestroy(() => sound.use(undefined));
</script>

<div class="hear">
  <div class="hear__bar">
    <SoundToggle volume />
    <SegmentedControl
      aria-label="Sound profile"
      bind:value={profile}
      items={[
        { value: 'system', label: 'System' },
        { value: 'glass', label: 'Glass' },
        { value: 'brutal', label: 'Brutalism' },
        { value: 'terminal', label: 'Terminal' },
      ]}
    />
  </div>
  <div class="hear__board" role="group" aria-label="Cue board">
    {#each CUES as cue (cue)}
      <Button variant="outlined" size="sm" sound={cue}>{cue}</Button>
    {/each}
  </div>
</div>

<style>
  .hear { display: grid; gap: var(--ca-space-4); }
  .hear__bar { display: flex; flex-wrap: wrap; align-items: center; gap: var(--ca-space-3); }
  .hear__board { display: flex; flex-wrap: wrap; gap: var(--ca-space-2); }
</style>
```
