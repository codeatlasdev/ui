# #5010 Inbox page

A complete inbox: ConversationItem rows in the master, a MessageBubble thread with a reply field in the detail, an empty state until a row is picked.

When to use: The messages screen of a CRM or helpdesk: two columns when wide, and on a phone the thread opens as a full-screen pane with Back.

## Facets

- Kind: component / layout
- Style: default
- Motion: subtle
- Interaction: press, keyboard
- Layout: split
- Color mode: both
- Density: regular
- Tech: css (cost: css)
- Reduced motion: handled; keyboard: operable
- Components: Avatar, ConversationItem, EmptyState, MasterDetail, MessageBubble, TextField
- Also searched as: inbox screen, chat screen, messages page

<script lang="ts">
  import { Avatar, ConversationItem, EmptyState, MasterDetail, MessageBubble, TextField } from '@codeatlasdev/ui';

  const m = {
    list: 'Conversations',
    empty: 'Select a conversation',
    compose: 'Message',
    composePlaceholder: 'Write a reply…',
  };

  type Thread = { id: string; name: string; time: string; unread: number; messages: { id: string; mine: boolean; text: string; time: string }[] };

  const threads: Thread[] = [
    {
      id: 'maria', name: 'Maria Souza', time: '14:32', unread: 2,
      messages: [
        { id: 'a', mine: false, text: 'Can you send the proposal today?', time: '14:30' },
        { id: 'b', mine: true, text: 'Sending it now.', time: '14:31' },
        { id: 'c', mine: false, text: 'Thanks, I will review it before 5.', time: '14:32' },
      ],
    },
    {
      id: 'joao', name: 'João Lima', time: '13:10', unread: 0,
      messages: [{ id: 'd', mine: false, text: 'Deal, thanks.', time: '13:10' }],
    },
    {
      id: 'ana', name: 'Ana Costa', time: 'Mon', unread: 0,
      messages: [{ id: 'e', mine: true, text: 'Invoice attached.', time: 'Mon' }],
    },
  ];

  let selected = $state<string | null>(null);
  const current = $derived(threads.find((t) => t.id === selected));
  const last = (t: Thread) => t.messages[t.messages.length - 1].text;
</script>

<div class="inbox-stage">
  <MasterDetail bind:selected detailTitle={current?.name ?? ''} listWidth="18rem" masterLabel={m.list}>
    {#snippet master()}
      {#each threads as t (t.id)}
        <ConversationItem title={t.name} time={t.time} preview={last(t)} unread={t.unread} active={selected === t.id} onclick={() => (selected = t.id)}>
          {#snippet avatar()}<Avatar alt={t.name} size="lg" />{/snippet}
        </ConversationItem>
      {/each}
    {/snippet}

    {#snippet detail()}
      {#if current}
        <div class="inbox-thread">
          <ul class="inbox-messages">
            {#each current.messages as msg (msg.id)}
              <li><MessageBubble direction={msg.mine ? 'out' : 'in'} time={msg.time}>{msg.text}</MessageBubble></li>
            {/each}
          </ul>
          <div class="inbox-composer"><TextField label={m.compose} placeholder={m.composePlaceholder} /></div>
        </div>
      {/if}
    {/snippet}

    {#snippet empty()}
      <EmptyState title={m.empty} />
    {/snippet}
  </MasterDetail>
</div>

<style>
  /* Demo frame only: in a route the page gives MasterDetail its height. */
  .inbox-stage { width: 100%; height: 26rem; border: var(--ca-border-width) solid var(--ca-outline-subtle); border-radius: var(--ca-radius-lg); overflow: hidden; }
  .inbox-thread { display: flex; flex-direction: column; min-height: 100%; }
  .inbox-messages { flex: 1; list-style: none; margin: 0; padding: var(--ca-space-4); display: flex; flex-direction: column; gap: var(--ca-space-2); }
  .inbox-composer { padding: var(--ca-space-3) var(--ca-space-4); }
</style>
