# #5027 Record page

A complete record page: a PageShell with a back link, an edit action and a summary aside, around a Card that holds a RecordList of the company's contacts.

When to use: The detail screen of one CRM record (company, account, project): facts beside the page, related records as dense rows that stack on a phone.

## Facets

- Kind: component / layout
- Style: default
- Motion: none
- Interaction: press, keyboard
- Layout: split
- Color mode: both
- Density: compact
- Tech: css (cost: css)
- Reduced motion: handled; keyboard: operable
- Components: Avatar, Button, Card, DropdownMenu, DropdownMenuItem, IconButton, PageShell, RecordCell, RecordList, RecordRow
- Also searched as: detail page, account page, company record

## Spec

- `<PageShell title={m.title} description={m.description} backHref="#companies" backLabel={m.back} asideLabel={m.aside}>`
- `<Card full flush title={m.contacts.title} headingLevel={2}>`
- `<RecordList columns="minmax(0,1.4fr) minmax(0,1fr) 6rem" actions labels={m.contacts.labels} aria-label={m.contacts.list}>`
- `<RecordCell label={m.contacts.role}>`

## Source

```svelte
<script lang="ts">
  import {
    Avatar, Button, Card, DropdownMenu, DropdownMenuItem, IconButton, PageShell, RecordCell, RecordList, RecordRow,
  } from '@codeatlasdev/ui';
  import { MoreHorizontal, Pencil } from '@lucide/svelte';

  const m = {
    back: 'Back to companies',
    title: 'Atlas Co',
    description: 'Customer since March 2026.',
    edit: 'Edit',
    aside: 'Company summary',
    summary: { title: 'Summary', owner: 'Owner', plan: 'Plan', since: 'Customer since' },
    contacts: { title: 'Contacts', list: 'Contacts', labels: ['Contact', 'Role', 'Last seen'], role: 'Role', seen: 'Last seen' },
    actions: (name: string) => `Actions for ${name}`,
    open: 'Open',
    remove: 'Remove',
  };

  const company = { owner: 'Ana Lima', plan: 'Business', since: 'Mar 14, 2026' };
  const contacts = [
    { id: 1, name: 'Maria Souza', role: 'Finance lead', seen: 'Today' },
    { id: 2, name: 'João Lima', role: 'Developer', seen: 'Yesterday' },
    { id: 3, name: 'Ana Costa', role: 'Procurement', seen: 'Oct 28' },
  ];
</script>

<div class="record-stage">
  <PageShell title={m.title} description={m.description} backHref="#companies" backLabel={m.back} asideLabel={m.aside}>
    {#snippet actions()}
      <Button variant="outlined" size="sm">
        {#snippet leading()}<Pencil size={14} />{/snippet}
        {m.edit}
      </Button>
    {/snippet}

    {#snippet aside()}
      <Card full title={m.summary.title} headingLevel={2}>
        <dl class="record-facts">
          <dt>{m.summary.owner}</dt><dd>{company.owner}</dd>
          <dt>{m.summary.plan}</dt><dd>{company.plan}</dd>
          <dt>{m.summary.since}</dt><dd>{company.since}</dd>
        </dl>
      </Card>
    {/snippet}

    <Card full flush title={m.contacts.title} headingLevel={2}>
      <RecordList columns="minmax(0,1.4fr) minmax(0,1fr) 6rem" actions labels={m.contacts.labels} aria-label={m.contacts.list}>
        {#each contacts as c (c.id)}
          <RecordRow>
            {#snippet identity()}
              <span class="record-who"><Avatar alt={c.name} size="sm" decorative /> <strong>{c.name}</strong></span>
            {/snippet}
            <RecordCell label={m.contacts.role}>{c.role}</RecordCell>
            <RecordCell label={m.contacts.seen}>{c.seen}</RecordCell>
            {#snippet actions()}
              <DropdownMenu align="end">
                {#snippet triggerChild(props)}
                  <IconButton aria-label={m.actions(c.name)} {...props}><MoreHorizontal size={18} /></IconButton>
                {/snippet}
                <DropdownMenuItem>{m.open}</DropdownMenuItem>
                <DropdownMenuItem>{m.remove}</DropdownMenuItem>
              </DropdownMenu>
            {/snippet}
          </RecordRow>
        {/each}
      </RecordList>
    </Card>
  </PageShell>
</div>
```
