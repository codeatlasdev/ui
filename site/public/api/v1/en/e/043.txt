Element: #043 States (inputs/select)
What it shows: Three selects showing a description, an error and a disabled state, with one disabled option.
When to use: Showing helper text, validation errors and locked fields on dropdowns in a form.
Spec:
- Plan select has `description="Enterprise is disabled."` and `bind:value={plan}`
- `plans` has Free, Pro, Enterprise, and Enterprise has `disabled: true`
- Region select has `error="Pick a region to continue."` and no bound value
- Size select has `disabled` with `bind:value={size}`
- 3 selects stacked in `.col` with `max-width: 320px`
Tags: select, error, disabled, description, validation
Needs: @codeatlasdev/ui@4.1.0
Components: Select
Live demo: https://ui.codeatlas.com.br/docs/inputs/select#states

<script lang="ts">
  import { Select } from '@codeatlasdev/ui';

  let size = $state('md');
  let plan = $state('free');

  const sizes = [
    { value: 'sm', label: 'Small' },
    { value: 'md', label: 'Medium' },
    { value: 'lg', label: 'Large' },
  ];

  const plans = [
    { value: 'free', label: 'Free' },
    { value: 'pro', label: 'Pro' },
    { value: 'enterprise', label: 'Enterprise', disabled: true },
  ];
</script>

<div class="col">
  <Select label="Plan" description="Enterprise is disabled." options={plans} bind:value={plan} />
  <Select label="Region" options={sizes} error="Pick a region to continue." />
  <Select label="Size" options={sizes} bind:value={size} disabled />
</div>

<style>
  .col {
    display: flex;
    flex-direction: column;
    gap: var(--ca-space-4);
    width: 100%;
    max-width: 320px;
  }
</style>
