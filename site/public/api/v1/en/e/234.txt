Element: #234 DocsNav and TableOfContents (docs-kit/docs)
What it shows: A DocsNav sidebar with two groups and a TableOfContents of four headings rendered side by side.
When to use: Documentation pages that need a section navigation tree next to an inline outline of the current page.
Spec:
- `<DocsNav {groups} pathname="/docs/docs-kit/docs/" onNavigate={() => {}} />`
- `groups` typed `DocsNavGroup[]` has 2 groups: Get started with 2 items and Messages with 4 items
- Messages items carry a `badge`: `badge: 'POST'`, `badge: 'GET'`, `badge: 'PUT'`, `badge: 'DELETE'`
- `<TableOfContents {headings} variant="inline" />`
- `headings` typed `DocsHeading[]` has 4 entries with `id`, `text` and `depth` of 2 or 3
Tags: docs-nav, table-of-contents, docs-kit, sidebar, navigation, headings
Needs: @codeatlasdev/ui@4.1.0
Components: DocsNav, TableOfContents
Live demo: https://ui.codeatlas.com.br/docs/docs-kit/docs#docsnav-and-tableofcontents

<script lang="ts">
  import { DocsNav, TableOfContents, type DocsHeading, type DocsNavGroup } from '@codeatlasdev/ui';

  const headings: DocsHeading[] = [
    { id: 'demo-prerequisites', text: 'Prerequisites', depth: 2 },
    { id: 'demo-auth', text: 'Authentication', depth: 2 },
    { id: 'demo-header', text: 'The x-api-key header', depth: 3 },
    { id: 'demo-send', text: 'Send a message', depth: 2 },
  ];

  const groups: DocsNavGroup[] = [
    {
      label: 'Get started',
      items: [
        { href: '/docs/docs-kit/docs/', title: 'Introduction' },
        { href: '/x/quick-start', title: 'Quick start' },
      ],
    },
    {
      label: 'Messages',
      items: [
        { href: '/x/send-text', title: 'Send text', badge: 'POST' },
        { href: '/x/list', title: 'List messages', badge: 'GET' },
        { href: '/x/edit', title: 'Edit message', badge: 'PUT' },
        { href: '/x/delete', title: 'Delete message', badge: 'DELETE' },
      ],
    },
  ];
</script>

<div class="pair">
  <DocsNav {groups} pathname="/docs/docs-kit/docs/" onNavigate={() => {}} />
  <TableOfContents {headings} variant="inline" />
</div>
