Element: #130 Streaming rows (display/live-list)
What it shows: A LiveList that streams in 40 simulated domain availability checks, one every 150ms, after a Start bulk check button is pressed.
When to use: Bulk jobs such as domain checks, imports or batch validations where results arrive one by one and the user wants to watch progress.
Spec:
- `<LiveList {items} total={TOTAL} {status} label="Domain checks" height="14rem">` with `const TOTAL = 40`
- State: `let items = $state<Check[]>([])` and `let status = $state<'idle' | 'running' | 'done'>('idle')`
- Row content rendered with `{#snippet item(row)}`: `row.domain` text plus a `Chip`
- `<Chip size="sm" variant={row.free ? 'success' : 'default'}` labelled Available or Taken from `row.free`
- `<Button variant="filled" onclick={start} disabled={status === 'running'}>` starts the stream, label becomes Checking… while running
- `start()` pushes one item with `domain` and `free` every 150ms with `setInterval` and sets `status = 'done'` at `TOTAL`
Tags: live-list, stream, progress, chip, bulk, status, real-time
Needs: @codeatlasdev/ui@4.1.0
Components: LiveList, Button, Chip
Live demo: https://ui.codeatlas.com.br/docs/display/live-list#streaming-rows

<script lang="ts">
  import { LiveList, Button, Chip } from '@codeatlasdev/ui';
  import { onDestroy } from 'svelte';

  type Check = { domain: string; free: boolean };
  const names = ['atlas', 'nimbus', 'orbit', 'prism', 'quill', 'ripple', 'sable', 'tundra', 'umbra', 'vertex', 'willow', 'zephyr'];
  const TOTAL = 40;

  let items = $state<Check[]>([]);
  let status = $state<'idle' | 'running' | 'done'>('idle');
  let timer: ReturnType<typeof setInterval> | undefined;

  function start() {
    items = [];
    status = 'running';
    clearInterval(timer);
    timer = setInterval(() => {
      const i = items.length;
      items.push({ domain: `${names[i % names.length]}${i}.com`, free: i % 3 !== 0 });
      if (items.length >= TOTAL) {
        clearInterval(timer);
        status = 'done';
      }
    }, 150);
  }
  onDestroy(() => clearInterval(timer));
</script>

<div class="wrap">
  <Button variant="filled" onclick={start} disabled={status === 'running'}>
    {status === 'running' ? 'Checking…' : 'Start bulk check'}
  </Button>
  <LiveList {items} total={TOTAL} {status} label="Domain checks" height="14rem">
    {#snippet item(row)}
      <span class="domain">{row.domain}</span>
      <Chip size="sm" variant={row.free ? 'success' : 'default'} label={row.free ? 'Available' : 'Taken'} />
    {/snippet}
  </LiveList>
</div>

<style>
  .wrap { display: grid; gap: var(--ca-space-3); width: 100%; }
  .domain { flex: 1; font-family: var(--ca-font-mono); }
</style>
