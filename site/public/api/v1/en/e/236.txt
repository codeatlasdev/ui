Element: #236 DocsSearch (docs-kit/docs)
What it shows: A DocsSearch dialog opened from an Open search button over a two-entry index, showing the selected href afterwards.
When to use: Docs site search opened by a button or shortcut that jumps to a page or heading.
Spec:
- `<Button variant="outlined" onclick={() => (open = true)}>` labelled Open search
- `<DocsSearch bind:open {index} onselect={(href) => (picked = href)} />`
- State: `let open = $state(false)` and `let picked = $state('')`
- `index` typed `DocsSearchEntry[]` has 2 entries (Send text, Webhooks) with `href`, `title`, `section`, `description`, `headings` and `text`
- `<span class="picked" aria-live="polite">` shows Selected plus the chosen href
Tags: docs-search, search, docs-kit, dialog, command-palette, index
Needs: @codeatlasdev/ui@4.1.0
Components: Button, DocsSearch
Live demo: https://ui.codeatlas.com.br/docs/docs-kit/docs#docssearch

<script lang="ts">
  import { Button, DocsSearch, type DocsSearchEntry } from '@codeatlasdev/ui';

  let open = $state(false);
  let picked = $state('');

  const index: DocsSearchEntry[] = [
    {
      href: '/x/send-text',
      title: 'Send text',
      section: 'API',
      description: 'Sends a text message to a number or group.',
      headings: [{ text: 'Request body', id: 'body' }],
      text: 'POST /messages/text with x-api-key and x-instance.',
    },
    {
      href: '/x/webhooks',
      title: 'Webhooks',
      section: 'Guide',
      description: 'Receive events in real time.',
      headings: [{ text: 'HMAC signature', id: 'hmac-signature' }],
      text: 'Retries with backoff and an HMAC signature.',
    },
  ];
</script>

<Button variant="outlined" onclick={() => (open = true)}>Open search</Button>
{#if picked}<span class="picked" aria-live="polite">Selected: <code>{picked}</code></span>{/if}
<DocsSearch bind:open {index} onselect={(href) => (picked = href)} />

<style>
  .picked { font-size: var(--ca-font-size-body2); color: var(--ca-text-secondary); }
  .picked code { font-family: var(--ca-font-mono); color: var(--ca-brand-text); }
</style>
