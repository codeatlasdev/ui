Element: #348 Spring switch (motion/micro)
What it shows: A switch whose thumb stretches to 1.3 on press and overshoots into place on release.
When to use: Settings switches where the thumb should feel physical rather than slide linearly.
Spec:
- `MotionStage` from `@codeatlasdev/ui` with `loop={2800}`
- `role="switch"` with `aria-checked={on}` and `aria-labelledby` pointing at the visible label
- `.sw__track--press .sw__thumb` is `transform: scaleX(1.3)` while pressed
- `.sw__track--on .sw__thumb` is `transform: translateX(20px)`
- `transition: transform 420ms cubic-bezier(0.175, 0.885, 0.32, 1.275)` overshoots the landing
- `.sw__track--still` sets `transition: none` under reduced motion
Motion:
- transform: translateX, scaleX (thumb): 420ms cubic-bezier(0.175, 0.885, 0.32, 1.275)
- background (track): 240ms cubic-bezier(0.2, 0, 0, 1)
Names: Spring switch | Elastic toggle | Switch elástico | Interrupteur élastique
Tags: motion, switch, toggle, spring, press, settings
Needs: @codeatlasdev/ui@4.3.0
Components: MotionStage
Live demo: https://ui.codeatlas.com.br/docs/motion/micro#spring-switch

<script lang="ts">
  import { MotionStage } from '@codeatlasdev/ui';

  const uid = $props.id();
  let on = $state(false);
  let pressed = $state(false);

  const flip = (node: HTMLElement) => {
    on = false;
    const t0 = setTimeout(() => (pressed = true), 500);
    const t1 = setTimeout(() => {
      pressed = false;
      on = true;
    }, 640);
    const t2 = setTimeout(() => (on = false), 1700);
    return () => {
      [t0, t1, t2].forEach(clearTimeout);
      pressed = false;
    };
  };
</script>

<MotionStage label="Spring switch: the thumb stretches on press and overshoots into place" loop={2800}>
  {#snippet children(m)}
    {#key m.cycle}
      <div class="sw" {@attach (node) => (m.playing && !m.reduced ? flip(node) : undefined)}>
        <button
          type="button"
          role="switch"
          aria-checked={on}
          aria-labelledby="{uid}-label"
          class="sw__track"
          class:sw__track--on={on}
          class:sw__track--press={pressed}
          class:sw__track--still={m.reduced}
          onclick={() => (on = !on)}
          onpointerdown={() => (pressed = true)}
          onpointerup={() => (pressed = false)}
          onpointerleave={() => (pressed = false)}
        >
          <span class="sw__thumb"></span>
        </button>
        <span class="sw__label" id="{uid}-label">Push notifications</span>
      </div>
    {/key}
  {/snippet}
</MotionStage>
