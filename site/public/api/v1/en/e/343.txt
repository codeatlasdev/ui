Element: #343 Like burst (motion/micro)
What it shows: A like button whose heart pops to 1.45 and throws eight particles outward while the count ticks up; aria-pressed carries the state.
When to use: Like, favorite or reaction buttons where a positive action deserves a small celebration.
Spec:
- `MotionStage` from `@codeatlasdev/ui` with `loop={2600}`
- `aria-pressed={liked}` on the button, count in `tabular-nums`
- `const PARTICLES = 8;` each `<i style="--a:{i * (360 / PARTICLES)}deg">` is one dot on a ring
- `@keyframes like-pop` scales the heart `1` to `1.45` and back
- `@keyframes like-particle` moves each dot `translateY(-26px)` while it fades to `opacity: 0` and `scale(0.3)`
- `{#key burst}` remounts the ring so the burst replays on every like; nothing bursts when reduced
Motion:
- transform: scale (heart): 460ms cubic-bezier(0.175, 0.885, 0.32, 1.275)
- transform, opacity (particles): 560ms cubic-bezier(0, 0, 0.2, 1)
Names: Like burst | Heart burst | Explosão de curtida | Explosion de cœur
Tags: motion, like, burst, particles, button, feedback
Needs: @codeatlasdev/ui@4.2.0
Components: MotionStage
Live demo: https://ui.codeatlas.com.br/docs/motion/micro#like-burst

<script lang="ts">
  import { MotionStage } from '@codeatlasdev/ui';
  import { Heart } from '@lucide/svelte';

  const PARTICLES = 8;
  let liked = $state(false);
  let count = $state(128);
  // Bumped on every like so the particle ring remounts and plays again.
  let burst = $state(0);

  function toggle() {
    liked = !liked;
    count += liked ? 1 : -1;
    if (liked) burst += 1;
  }

  // Autoplay likes, then unlikes, once per cycle.
  const autoplay = (node: HTMLElement) => {
    liked = false;
    count = 128;
    const t1 = setTimeout(toggle, 600);
    const t2 = setTimeout(() => {
      if (liked) toggle();
    }, 1700);
    return () => {
      clearTimeout(t1);
      clearTimeout(t2);
    };
  };
</script>

<MotionStage label="Like burst: a heart pops and throws eight particles" loop={2600}>
  {#snippet children(m)}
    {#key m.cycle}
      <div class="like" {@attach (node) => (m.playing && !m.reduced ? autoplay(node) : undefined)}>
        <button
          type="button"
          class="like__btn"
          class:like__btn--on={liked}
          aria-pressed={liked}
          onclick={toggle}
        >
          <span class="like__heart" class:like__heart--pop={liked && !m.reduced}>
            <Heart size={22} aria-hidden="true" fill={liked ? 'currentColor' : 'none'} />
          </span>
          {#if liked && !m.reduced}
            {#key burst}
              <span class="like__ring" aria-hidden="true">
                {#each Array.from({ length: PARTICLES }, (_, i) => i) as i}
                  <i style="--a:{i * (360 / PARTICLES)}deg"></i>
                {/each}
              </span>
            {/key}
          {/if}
          <span class="like__label">Like</span>
          <span class="like__count">{count}</span>
        </button>
      </div>
    {/key}
  {/snippet}
</MotionStage>
