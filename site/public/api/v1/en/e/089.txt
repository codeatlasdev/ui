Element: #089 Right (default) (feedback/drawer)
What it shows: A default drawer titled Filters with status and role filter groups, two switches and Reset and Apply actions.
When to use: Filter panels for tables and lists that would crowd the page if shown inline.
Spec:
- `<Drawer bind:open title="Filters">` with no side or width prop
- Two `<FilterGroup` blocks: status with 2 options (Active, Inactive) and role with 3 options (Admin, Editor, Viewer)
- Filter state: `onchange={(v) => (statusF = v)}` and `onchange={(v) => (roleF = v)}`, both starting at `'all'`
- Two `<Switch label="Show inactive only" />` and `<Switch label="Include archived" />` inside `<Stack gap="4">`
- `{#snippet actions()}` holds ghost Reset and filled Apply buttons
Tags: drawer, filters, filter-group, switch, side-panel
Needs: @codeatlasdev/ui@4.1.0
Components: Drawer, Button, Switch, Stack, FilterGroup
Live demo: https://ui.codeatlas.com.br/docs/feedback/drawer#right-default

<script lang="ts">
  import { Drawer, Button, Switch, Stack, FilterGroup } from '@codeatlasdev/ui';
  import { SlidersHorizontal } from '@lucide/svelte';

  let open = $state(false);
  let statusF = $state('all');
  let roleF = $state('all');
  const statusOpts = [{ value: 'active', label: 'Active' }, { value: 'inactive', label: 'Inactive' }];
  const roleOpts = [{ value: 'Admin', label: 'Admin' }, { value: 'Editor', label: 'Editor' }, { value: 'Viewer', label: 'Viewer' }];
</script>

<Button variant="outlined" onclick={() => (open = true)}>
  {#snippet leading()}<SlidersHorizontal size={14} />{/snippet}
  Open drawer
</Button>

<Drawer bind:open title="Filters">
  <Stack gap="4">
    <div>
      <p class="label">Status</p>
      <FilterGroup options={statusOpts} value={statusF} onchange={(v) => (statusF = v)} />
    </div>
    <div>
      <p class="label">Role</p>
      <FilterGroup options={roleOpts} value={roleF} onchange={(v) => (roleF = v)} />
    </div>
    <Switch label="Show inactive only" />
    <Switch label="Include archived" />
  </Stack>
  {#snippet actions()}
    <Button variant="ghost" onclick={() => (open = false)}>Reset</Button>
    <Button variant="filled" onclick={() => (open = false)}>Apply</Button>
  {/snippet}
</Drawer>

<style>
  .label {
    font-size: 12px;
    font-weight: 500;
    color: var(--ca-text-secondary);
    margin: 0 0 8px;
    text-transform: uppercase;
    letter-spacing: 0.05em;
  }
</style>
