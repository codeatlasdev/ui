Element: #431 Board (foundation/drag-and-drop)
What it shows: A three-column board: cards reorder inside a column and move between columns by mouse, touch or keyboard.
When to use: Kanban boards, status pipelines and any set of lists whose items move from one to another.
Spec:
- Every column is `{@attach dnd.sortable(() => ({ id: col, group: 'board',`; lists with the same `group` accept each other's cards
- `onDrop` receives the moved id with its `from` and `to` positions and applies it with `applyMove(lists, e, (c) => c.id)`
- Keyboard: Space lifts the card, ArrowUp and ArrowDown move it inside the column, ArrowLeft and ArrowRight jump to the next column, Enter drops, Escape cancels
- Each card is `<li data-dnd-item={card.id}` with a focusable `button`
Names: Kanban board | Drag between lists | Cards between columns | Quadro kanban | Tablero kanban
Tags: drag, drop, kanban, board, move, lists
Needs: @codeatlasdev/ui@4.3.0
Components: DndLayer
Live demo: https://ui.codeatlas.com.br/docs/foundation/drag-and-drop#board

<script lang="ts">
  import { applyMove, DndLayer, useDnd, type DndDropEvent } from '@codeatlasdev/ui';

  type Card = { id: string; title: string };
  const names: Record<string, string> = { todo: 'To do', doing: 'In progress', done: 'Done' };
  let lists = $state<Record<string, Card[]>>({
    todo: [{ id: 'c1', title: 'Design the empty state' }, { id: 'c2', title: 'Audit contrast' }],
    doing: [{ id: 'c3', title: 'Wire the export menu' }],
    done: [{ id: 'c4', title: 'Ship dark mode' }],
  });

  const dnd = useDnd();

  function onDrop(e: DndDropEvent) {
    lists = applyMove(lists, e, (c) => c.id);
  }
  const title = (id: string) => Object.values(lists).flat().find((c) => c.id === id)?.title ?? id;
</script>

<DndLayer {dnd} />
<div class="bd">
  {#each Object.keys(lists) as col (col)}
    <section class="bd__col" aria-labelledby="bd-{col}">
      <h4 id="bd-{col}" class="bd__h">{names[col]} <span>{lists[col].length}</span></h4>
      <ul
        class="bd__list"
        aria-labelledby="bd-{col}"
        {@attach dnd.sortable(() => ({ id: col, group: 'board', label: names[col], getItems: () => lists[col].map((c) => c.id), getLabel: title, onDrop }))}
      >
        {#each lists[col] as card (card.id)}
          <li data-dnd-item={card.id} class="bd__card">
            <button type="button" class="bd__btn" aria-describedby={dnd.instructionsId}>{card.title}</button>
          </li>
        {/each}
      </ul>
    </section>
  {/each}
</div>
