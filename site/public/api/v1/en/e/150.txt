Element: #150 longpress (navigation/actions)
What it shows: Dispatches a longpress event (listen with `onlongpress`) after the pointer is held for the given duration.
Tags: longpress
Needs: @codeatlasdev/ui@4.0.0
Components: none
Live demo: https://ui.codeatlas.com.br/docs/navigation/actions#longpress

<script lang="ts">
  import { longpress } from '@codeatlasdev/ui/actions';

  let fired = $state(false);
  function done() {
    fired = true;
    setTimeout(() => (fired = false), 1500);
  }
</script>

<button use:longpress={800} onlongpress={done} class="btn">
  {fired ? 'Long press!' : 'Press and hold (800ms)'}
</button>
