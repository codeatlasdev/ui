Element: #402 Context menu (navigation/menu)
What it shows: A dashed button area that opens a Rename, Duplicate, Delete menu at the pointer on right-click, or from the keyboard with Shift+F10 or the ContextMenu key.
When to use: Context menus on rows, files or canvases: the menu opens where the user pointed and moves when they right-click again.
Spec:
- `let open = $state(false)` and `let point = $state<{ x: number; y: number } | null>(null)`
- The area is a native `<button type="button" class="area" {oncontextmenu} {onkeydown}>` so it is focusable and named by its visible text
- `oncontextmenu` calls `e.preventDefault()`, sets `point = { x: e.clientX, y: e.clientY }` and `open = true`
- `onkeydown` handles `ContextMenu` and Shift+F10 and anchors the menu to the area's top-left
- `<DropdownMenu bind:open anchor={point}>` has no trigger; items Rename, Duplicate and a destructive Delete after `<DropdownMenuSeparator />`
- Each item calls `function pick(action: string)`, which sets `last = action` and calls `announce`; `Last action: {last}` is shown below
Tags: dropdown-menu, context-menu, right-click, anchor, bind-open, keyboard
Needs: @codeatlasdev/ui@4.3.0
Components: DropdownMenu, DropdownMenuItem, DropdownMenuSeparator
Live demo: https://ui.codeatlas.com.br/docs/navigation/menu#context-menu

<script lang="ts">
  import { announce, DropdownMenu, DropdownMenuItem, DropdownMenuSeparator } from '@codeatlasdev/ui';

  let open = $state(false);
  let point = $state<{ x: number; y: number } | null>(null);
  let last = $state('');

  function pick(action: string) {
    last = action;
    announce(`Last action: ${action}`);
  }

  function oncontextmenu(e: MouseEvent) {
    e.preventDefault();
    point = { x: e.clientX, y: e.clientY };
    open = true;
  }

  function onkeydown(e: KeyboardEvent) {
    if (e.key !== 'ContextMenu' && !(e.shiftKey && e.key === 'F10')) return;
    e.preventDefault();
    const box = e.currentTarget.getBoundingClientRect();
    point = { x: box.left + 16, y: box.top + 16 };
    open = true;
  }
</script>

<button type="button" class="area" {oncontextmenu} {onkeydown}>
  Right-click here, or focus it and press Shift+F10.
</button>

<DropdownMenu bind:open anchor={point}>
  <DropdownMenuItem onclick={() => pick('Rename')}>Rename</DropdownMenuItem>
  <DropdownMenuItem onclick={() => pick('Duplicate')}>Duplicate</DropdownMenuItem>
  <DropdownMenuSeparator />
  <DropdownMenuItem variant="destructive" onclick={() => pick('Delete')}>Delete</DropdownMenuItem>
</DropdownMenu>

{#if last}<span class="note">Last action: {last}</span>{/if}
