Element: #561 Short panel (layout/app-shell)
What it shows: An app shell with one module of two items and collapseShortPanel: only the rail shows, the page uses the full width.
When to use: Small modules (settings, profile) where a 200px panel would hold two links and leave the rest empty.
Spec:
- `<AppShellLayout {modules} {pathname} collapseShortPanel persistPanel={false} mainId={null} skipLink={false}>` in a 320px tall frame
- One module `id: 'settings'` (Settings icon) with two items: Profile and Billing
- Page is `<PageShell title="Profile" description="A two-item module needs no panel: the rail is enough.">` with a glass `Card` hint
Tags: app-shell, panel, collapse, rail, sidebar, navigation
Needs: @codeatlasdev/ui@4.5.0
Components: AppShellLayout, PageShell, Card
Live demo: https://ui.codeatlas.com.br/docs/layout/app-shell#short-panel

<script lang="ts">
  import { AppShellLayout, PageShell, Card, type NavModule } from '@codeatlasdev/ui';
  import { Settings, User, CreditCard } from '@lucide/svelte';

  const modules: NavModule[] = [
    {
      id: 'settings',
      label: 'Settings',
      icon: Settings,
      basePath: '/settings',
      items: [
        { id: 'profile', label: 'Profile', icon: User, path: '/settings/profile' },
        { id: 'billing', label: 'Billing', icon: CreditCard, path: '/settings/billing' },
      ],
    },
  ];

  let pathname = $state('/settings/profile');

  function intercept(e: MouseEvent) {
    if (e.defaultPrevented || e.button !== 0 || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey) return;
    const a = (e.target as Element).closest('a[href^="/"]');
    if (!a) return;
    e.preventDefault();
    pathname = a.getAttribute('href')!;
  }
</script>

<!-- svelte-ignore a11y_no_static_element_interactions, a11y_click_events_have_key_events -->
<div class="stage" onclick={intercept}>
  <AppShellLayout {modules} {pathname} collapseShortPanel persistPanel={false} mainId={null} skipLink={false}>
    <PageShell title="Profile" description="A two-item module needs no panel: the rail is enough.">
      <Card material="glass" full>
        <p class="hint">With collapseShortPanel the content takes the width a two-link panel would waste.</p>
      </Card>
    </PageShell>
  </AppShellLayout>
</div>

<style>

  }
</style>
