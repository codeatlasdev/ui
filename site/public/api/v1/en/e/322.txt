Element: #322 Parallax (motion/scroll)
What it shows: Three layers (sky, ridge, foreground) plus a caption move at different speeds while the container scrolls, so the scene gains depth. Native scroll timeline, with a scroll listener fallback.
When to use: Hero images, editorial headers, product storytelling: any pinned scene where background should move slower than foreground.
Spec:
- `MotionStage` from `@codeatlasdev/ui` with `loop={6200}`: autoplay scrolls the container, a wheel or touch scroll takes over
- A sticky scene inside a taller scroller: `position: sticky;` on `.scene__sticky` and `block-size: 26rem;` on `.scene__track`
- Native: `animation-timeline: --scene` over `scroll-timeline: --scene block`, inside `@supports (animation-timeline: scroll())`
- Layer travel at full scroll: sky `-6%`, ridge `-22%`, foreground `-48%`, caption `30%` while fading to opacity `0.15`
- Fallback without support: a `scroll` listener writes progress to `--p` and the transforms read `translateY(calc(var(--p) * -22%))`
- Under reduced motion `--p` is `1` and nothing moves (`.scene--final`)
Motion:
- transform: translateY, sky layer to -6%: animation-timeline: --scene linear
- transform: translateY, ridge layer to -22%: animation-timeline: --scene linear
- transform: translateY, foreground layer to -48%: animation-timeline: --scene linear
- transform: translateY to 30% and opacity to 0.15, caption: animation-timeline: --scene linear

<script lang="ts">
  import { MotionStage, type MotionCtx } from '@codeatlasdev/ui';
  import type { Attachment } from 'svelte/attachments';

  // Scroll progress 0..1 of the container. With CSS scroll-driven animations the browser computes it;
  // elsewhere the scroll listener writes it to --p and the same transforms read it.
  const native = typeof CSS !== 'undefined' && CSS.supports('animation-timeline: scroll()');

  const track = (m: MotionCtx): Attachment<HTMLElement> => (el) => {
    if (native) return;
    const write = () => {
      const max = el.scrollHeight - el.clientHeight;
      if (!m.reduced) el.style.setProperty('--p', String(max > 0 ? el.scrollTop / max : 0));
    };
    write();
    el.addEventListener('scroll', write, { passive: true });
    return () => el.removeEventListener('scroll', write);
  };

  // Autoplay = scroll down in 3000ms, once per cycle; wheel or touch = takeover.
  const drive = (m: MotionCtx): Attachment<HTMLElement> => (el) => {
    const stop = () => m.takeover();
    el.addEventListener('wheel', stop, { passive: true });
    el.addEventListener('touchmove', stop, { passive: true });
    let raf = 0;
    if (m.playing) {
      void m.cycle;
      el.scrollTop = 0;
      const to = el.scrollHeight - el.clientHeight;
      const t0 = performance.now();
      const step = (now: number) => {
        if (!m.playing) return; // taken over or paused: never write a frame after the user starts scrolling
        const t = Math.min(1, (now - t0) / 3000);
        el.scrollTop = to * (t < 0.5 ? 4 * t * t * t : 1 - (-2 * t + 2) ** 3 / 2);
        if (t < 1) raf = requestAnimationFrame(step);
      };
      raf = requestAnimationFrame(step);
    }
    return () => {
      cancelAnimationFrame(raf);
      el.removeEventListener('wheel', stop);
      el.removeEventListener('touchmove', stop);
    };
  };
</script>
