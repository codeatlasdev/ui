Element: #020 Basic (inputs/file-upload)
What it shows: A multi-file FileUpload with a size limit and hint that reports how many files are selected.
When to use: Attachment and document upload areas that accept several files and cap their size.
Spec:
- `bind:files` with `let files = $state<File[]>([])`
- `multiple`
- `hint="PNG, JPG, PDF up to 10 MB"`
- `maxSize={10 * 1024 * 1024}`
- `{#if files.length > 0}` shows the selected file count with a plural `s`
- Wrapper `.wrap` has `max-width: 480px`
Tags: file-upload, dropzone, multiple, max-size, files, attachment
Needs: @codeatlasdev/ui@4.1.0
Components: FileUpload
Live demo: https://ui.codeatlas.com.br/docs/inputs/file-upload#basic

<script lang="ts">
  import { FileUpload } from '@codeatlasdev/ui';

  let files = $state<File[]>([]);
</script>

<div class="wrap">
  <FileUpload
    bind:files
    multiple
    hint="PNG, JPG, PDF up to 10 MB"
    maxSize={10 * 1024 * 1024}
  />
  {#if files.length > 0}
    <p class="out">{files.length} file{files.length > 1 ? 's' : ''} selected</p>
  {/if}
</div>

<style>
  .wrap {
    width: 100%;
    max-width: 480px;
  }

  .out {
    margin: var(--ca-space-2) 0 0;
    font-size: var(--ca-font-size-body2);
    color: var(--ca-text-secondary);
  }
</style>
