Element: #304 Typewriter (motion/text)
What it shows: A prompt that types itself at 55ms per character, holds for 1100ms, then deletes at 28ms per character while a caret blinks. Typing in the field below takes over the line.
When to use: AI and search inputs with example prompts, hero taglines that cycle phrases, terminal and chat demos and onboarding hints.
Spec:
- `MotionStage` from `@codeatlasdev/ui` with `label="Typewriter: a prompt types itself, holds, then deletes"` and `loop={4800}`
- `const TYPE = 55`, `const ERASE = 28` and `const HOLD = 1100` are the typing, deleting and hold times in milliseconds
- One autoplay cycle types `phrases[m.cycle % phrases.length]`, holds it, then deletes it
- `animation: tw-blink 1060ms steps(1, end) infinite` blinks the caret with a hard on/off step
- The typed text and caret are `aria-hidden="true"`; the labelled `id="tw-input"` input is the accessible control
- `bind:value={draft}`: typing in the field replaces the autoplay text, the manual takeover
- Under reduced motion `class:tw__caret--blink={!m.reduced}` stops the blink and the first phrase is shown complete
Motion:
- text content (characters typed, held, then deleted): 55ms per character linear delay 1100ms
- opacity (caret blink): 1060ms steps(1, end)
Names: Typewriter | Typing effect | Type and delete | Máquina de escrever | Efeito de digitação | Machine à écrire

<script lang="ts">
  import { MotionStage, type MotionCtx } from '@codeatlasdev/ui';

  const phrases = ['Invoice #4821 is paid.', 'Reconcile 312 transactions.', 'Ask anything about Q3.'];
  const TYPE = 55; // 55ms per character, linear (a constant pace)
  const ERASE = 28; // 28ms per deleted character
  const HOLD = 1100; // 1100ms pause on a finished phrase

  let shown = $state(phrases[0]);
  let draft = $state('');

  /** One autoplay cycle types a phrase, holds it, then deletes it; the next cycle picks the next phrase. */
  const autoplay = (m: MotionCtx) => () => {
    if (m.reduced) {
      shown = phrases[0];
      return;
    }
    const phrase = phrases[m.cycle % phrases.length];
    let i = 0;
    let dir = 1;
    let timer: ReturnType<typeof setTimeout>;
    const step = () => {
      i += dir;
      shown = phrase.slice(0, i);
      if (dir === 1 && i === phrase.length) {
        dir = -1;
        timer = setTimeout(step, HOLD);
      } else if (dir === -1 && i === 0) {
        return;
      } else timer = setTimeout(step, dir === 1 ? TYPE : ERASE);
    };
    shown = '';
    timer = setTimeout(step, TYPE);
    return () => clearTimeout(timer);
  };
</script>

<MotionStage label="Typewriter: a prompt types itself, holds, then deletes" loop={4800}>
  {#snippet children(m)}
    <div class="tw" {@attach autoplay(m)}>
      <label class="tw__label" for="tw-input">Assistant prompt</label>
      <div class="tw__box">
        <!-- The finished phrase is announced once by the input below; the typing is decoration. -->
        <span class="tw__text" aria-hidden="true">{m.playing || m.reduced ? shown : draft || shown}</span><span
          class="tw__caret"
          class:tw__caret--blink={!m.reduced}
          aria-hidden="true"
        ></span>
      </div>
      <input
        id="tw-input"
        class="tw__input"
        placeholder="Type to take over"
        autocomplete="off"
        bind:value={draft}
      />
    </div>
  {/snippet}
</MotionStage>
