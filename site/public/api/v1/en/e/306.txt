Element: #306 Count up (motion/text)
What it shows: A KPI that counts from 0 to 12,480 in 1400ms with an ease-out-expo curve, so the last digits settle slowly. A Count again button replays it on demand.
When to use: Stat blocks and metrics sections that scroll into view, totals after a report loads, and any headline number worth a moment of attention.
Spec:
- `MotionStage` from `@codeatlasdev/ui` with `label="Count up: a number counts from zero and eases into its total"` and `loop={3200}`
- `const DURATION = 1400` with `1 - 2 ** (-10 * t)` as the easeOutExpo curve
- `n = Math.round(target * ease(t))` on every `requestAnimationFrame`, `toLocaleString('en-US')` formats the digits
- `role="img"` with `aria-label` states the final total once; the counting `<span aria-hidden="true">` is decoration
- The `Button` `onclick` calls `run()`: pressing Count again is the takeover and re-runs the count on demand
- Under reduced motion the attachment sets `n = target` and no frame loop starts
Motion:
- text content (integer counting from 0 to the total): 1400ms 1 - 2 ** (-10 * t)
Names: Count up | Number counter | Animated counter | Contagem progressiva | Contador animado | Compteur animé

<script lang="ts">
  import { Button, MotionStage, type MotionCtx } from '@codeatlasdev/ui';

  const target = 12480;
  const DURATION = 1400; // 1400ms count, eased out

  let n = $state(target);
  let raf = 0;

  /** easeOutExpo: fast start, long settle, so the last digits tick slowly into place. */
  const ease = (t: number) => (t >= 1 ? 1 : 1 - 2 ** (-10 * t));

  function run() {
    cancelAnimationFrame(raf);
    const start = performance.now();
    const frame = (now: number) => {
      const t = Math.min(1, (now - start) / DURATION);
      n = Math.round(target * ease(t));
      if (t < 1) raf = requestAnimationFrame(frame);
    };
    n = 0;
    raf = requestAnimationFrame(frame);
  }

  const autoplay = (m: MotionCtx) => () => {
    void m.cycle;
    if (m.reduced) {
      n = target;
      return;
    }
    run();
    return () => cancelAnimationFrame(raf);
  };
</script>

<MotionStage label="Count up: a number counts from zero and eases into its total" loop={3200}>
  {#snippet children(m)}
    <div class="cu" {@attach autoplay(m)}>
      <p class="cu__label">Active customers</p>
      <!-- The final number is the accessible value; the counting digits are decoration. -->
      <p class="cu__value" role="img" aria-label={`${target.toLocaleString('en-US')} active customers`}>
        <span aria-hidden="true">{n.toLocaleString('en-US')}</span>
      </p>
      <p class="cu__delta">+18.4% since last quarter</p>
      <Button size="sm" variant="outlined" onclick={() => !m.reduced && run()}>Count again</Button>
    </div>
  {/snippet}
</MotionStage>
