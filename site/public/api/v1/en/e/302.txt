Element: #302 Odometer roll (motion/text)
What it shows: A KPI value made of digit columns that slide to the new digit, the ones place first and each column 60ms after the last. It cycles through values on its own; typing a number rolls it.
When to use: Dashboards, revenue and counter KPIs, price changes, scores and any number that updates in place and should feel like a mechanical counter.
Spec:
- `MotionStage` from `@codeatlasdev/ui` with `label="Odometer: a KPI value rolls digit by digit"` and `loop={2600}`
- Each digit is a `.odo__col` window (`overflow: hidden`, `block-size: 1em`) holding a strip of `0` to `9`
- `transform: translateY(calc(var(--d) * -1em))` slides the strip to the digit `--d`
- `transition: transform 900ms cubic-bezier(0.2, 0.9, 0.2, 1) calc(var(--i) * 60ms)` with `--i` counted from the right, so the ones place rolls first
- `role="img"` with `aria-label` carries the whole number once; the columns are `aria-hidden="true"`
- `oninput={onType}` sets `value`: typing a number in the field is the takeover and rolls the columns
- `{@attach autoplay(m)}` walks `values[cycle % values.length]` only while `playing`; under reduced motion `class:odo__strip--run={!m.reduced}` drops the transition and the last value is shown
Motion:
- transform: translateY (digit strip inside an overflow window): 900ms cubic-bezier(0.2, 0.9, 0.2, 1) stagger calc(var(--i) * 60ms)

<script lang="ts">
  import { MotionStage, type MotionCtx } from '@codeatlasdev/ui';

  const values = [48210, 51904, 49377, 56012];
  const digits = '0123456789'.split('');

  let value = $state(values[0]);
  let typed = $state('48210');

  /** Autoplay walks through the KPI values only while playing; reduced motion pins the last one. */
  const autoplay = (m: MotionCtx) => () => {
    const { cycle, reduced, playing } = m;
    if (reduced) value = values[values.length - 1];
    else if (playing) value = values[cycle % values.length];
    else return;
    typed = String(value);
  };

  function onType(e: Event) {
    typed = (e.currentTarget as HTMLInputElement).value.replace(/\D/g, '').slice(0, 7);
    value = Number(typed || 0);
  }

  // Each digit is a column of 0-9; the column slides to the digit with translateY.
  const cols = $derived(String(value).split('').map(Number));
</script>

<MotionStage label="Odometer: a KPI value rolls digit by digit" loop={2600}>
  {#snippet children(m)}
    <div class="odo" {@attach autoplay(m)}>
      <p class="odo__label">Monthly revenue (USD)</p>
      <!-- One stable value for assistive tech; the rolling columns are decoration. -->
      <p class="odo__value" role="img" aria-label={`${value.toLocaleString('en-US')} dollars`}>
        <span aria-hidden="true">$</span>
        {#each cols as d, i (cols.length - i)}
          <span class="odo__col" aria-hidden="true">
            <span class="odo__strip" class:odo__strip--run={!m.reduced} style="--d:{d}; --i:{cols.length - 1 - i}">
              {#each digits as n (n)}<span class="odo__digit">{n}</span>{/each}
            </span>
          </span>
        {/each}
      </p>
      <label class="odo__field">
        <span class="odo__hint">Type a number to roll it</span>
        <input
          class="odo__input"
          inputmode="numeric"
          autocomplete="off"
          value={typed}
          oninput={onType}
        />
      </label>
    </div>
  {/snippet}
</MotionStage>
