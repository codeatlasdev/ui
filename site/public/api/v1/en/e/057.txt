Element: #057 With icons (inputs/text-field)
What it shows: Search field with a leading icon and a password field with a trailing show/hide button.
When to use: Search boxes and password inputs that need an inline icon or action.
Spec:
- Search field uses `{#snippet leading()}` with `<Search size={16} stroke-width={2} />`
- Password field uses `{#snippet trailing()}` containing a `<button type="button" class="reveal"`
- Field `type={reveal ? 'text' : 'password'}` toggled by `onclick={() => (reveal = !reveal)}`
- Button sets `aria-pressed={reveal}` and `aria-label={reveal ? 'Hide password' : 'Show password'}`
- Icons `Eye` and `EyeOff` come from `@lucide/svelte`
Tags: text-field, icon, leading, trailing, password, search, snippet
Needs: @codeatlasdev/ui@4.1.0
Components: TextField
Live demo: https://ui.codeatlas.com.br/docs/inputs/text-field#with-icons

<script lang="ts">
  import { TextField } from '@codeatlasdev/ui';
  import { Eye, EyeOff, Search } from '@lucide/svelte';

  let reveal = $state(false);
  let password = $state('');
</script>

<div class="col">
  <TextField label="Search" placeholder="Type to search…">
    {#snippet leading()}<Search size={16} stroke-width={2} />{/snippet}
  </TextField>
  <TextField label="Password" type={reveal ? 'text' : 'password'} placeholder="••••••••" bind:value={password}>
    {#snippet trailing()}
      <button type="button" class="reveal" aria-label={reveal ? 'Hide password' : 'Show password'} aria-pressed={reveal} onclick={() => (reveal = !reveal)}>
        {#if reveal}<EyeOff size={16} stroke-width={2} />{:else}<Eye size={16} stroke-width={2} />{/if}
      </button>
    {/snippet}
  </TextField>
</div>

<style>
  .col {
    display: flex;
    flex-direction: column;

  }

    min-width: var(--ca-touch-target);
    min-height: var(--ca-touch-target);
    margin-block: calc((var(--ca-touch-target) - 1lh) / -2);
    display: inline-flex;
    align-items: center;
    justify-content: center;
    display: inline-flex;
    padding: 0;
    border: 0;
    background: none;
    color: inherit;
    cursor: pointer;
  }
</style>
