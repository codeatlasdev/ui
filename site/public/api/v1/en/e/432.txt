Element: #432 Drop zones (foundation/drag-and-drop)
What it shows: Task chips dragged from a tray onto weekday drop zones, with a dashed highlight while a chip is over a day.
When to use: Scheduling, assigning and filing: any place a loose item is dropped onto a target that is not a sortable list.
Spec:
- Each chip is `{@attach dnd.draggable(() => ({ id: t.id, label: t.title }))}` on an element with `data-dnd-item`
- Each day is `{@attach dnd.dropzone(() => ({ id: day, label: day, onDrop: (e) => drop(day, e) }))}`; the zone is outlined while a chip is over it
- Keyboard: Space lifts a chip, the arrows pick the day, Enter drops; the zone name is announced to screen readers
Names: Drop zones | Drag onto a day | Calendar drop target | Zonas de soltar | Zonas para soltar
Tags: drag, drop, dropzone, calendar, schedule
Needs: @codeatlasdev/ui@4.3.0
Components: DndLayer
Live demo: https://ui.codeatlas.com.br/docs/foundation/drag-and-drop#drop-zones

<script lang="ts">
  import { DndLayer, useDnd, type DndDropEvent } from '@codeatlasdev/ui';

  const days = ['Mon', 'Tue', 'Wed', 'Thu'];
  const tasks = [
    { id: 'standup', title: 'Stand-up' },
    { id: 'review', title: 'Design review' },
    { id: 'demo', title: 'Demo prep' },
  ];
  let placed = $state<Record<string, string>>({});

  const dnd = useDnd();

  function drop(day: string, e: DndDropEvent) {
    placed = { ...placed, [e.id]: day };
  }
  const unplaced = $derived(tasks.filter((t) => !placed[t.id]));
</script>

<DndLayer {dnd} />
<div class="dz">
  <ul class="dz__tray" aria-label="Unscheduled">
    {#each unplaced as t (t.id)}
      <li class="dz__chip" data-dnd-item={t.id} {@attach dnd.draggable(() => ({ id: t.id, label: t.title }))}>
        <button type="button" class="dz__btn" aria-describedby={dnd.instructionsId}>{t.title}</button>
      </li>
    {/each}
    {#if unplaced.length === 0}<li class="dz__empty">Everything is scheduled</li>{/if}
  </ul>
  <div class="dz__days">
    {#each days as day (day)}
      <div class="dz__day" role="group" aria-label={day} {@attach dnd.dropzone(() => ({ id: day, label: day, onDrop: (e) => drop(day, e) }))}>
        <strong>{day}</strong>
        <ul class="dz__placed">
          {#each tasks.filter((t) => placed[t.id] === day) as t (t.id)}<li>{t.title}</li>{/each}
        </ul>
      </div>
    {/each}
  </div>
</div>
