Element: #800 Themed map (map/stage)
What it shows: A map stage with an offline light and dark basemap, four deck.gl store circles in the brand color and zoom and compass controls.
When to use: A map-first screen whose basemap and data layers must follow the app's light and dark theme.
Spec:
- `<MapStage` with `{styles}`, `{layers}`, `label="Store map"`, `padding={{ top: 24, right: 24, bottom: 24, left: 24 }}` and a `fit` of the southeast Brazil bounds
- `styles` is a `$derived.by` whose `light` and `dark` entries are one inline style with a single `background` layer painted with `rgbaCss(mapColors().surface)`, so the demo needs no tiles, token or hex color
- `layers` is a `$derived.by` that calls `mapColors()` and returns a deck.gl `ScatterplotLayer` with `getFillColor: c.brandMuted` and `getLineColor: c.brand`, so it re-themes on scheme or preset change
- `<MapControls />` as the child of `MapStage`
- The stage box is `360px` high with `var(--ca-radius-lg)` corners
Tags: map, deck.gl, maplibre, theme, dark-mode, controls
Needs: @codeatlasdev/ui@4.5.0 + maplibre-gl
Components: MapControls, MapStage
Live demo: https://ui.codeatlas.com.br/docs/map/stage#themed-map

<script lang="ts">
  import { ScatterplotLayer } from '@deck.gl/layers';
  import { MapControls, MapStage, mapColors, rgbaCss } from '@codeatlasdev/ui/map';

  // Offline basemap: one background layer painted with the surface token, so no tiles, no token and
  // no hex. `mapColors()` reads the theme, so the style is rebuilt when the scheme or preset changes.
  const styles = $derived.by(() => {
    const background = rgbaCss(mapColors().surface);
    const style = {
      version: 8 as const,
      sources: {},
      layers: [{ id: 'bg', type: 'background' as const, paint: { 'background-color': background } }],
    };
    return { light: style, dark: style };
  });

  const stores = [
    { name: 'Sao Paulo', coordinates: [-46.63, -23.55], sales: 90 },
    { name: 'Rio', coordinates: [-43.17, -22.91], sales: 70 },
    { name: 'Belo Horizonte', coordinates: [-43.94, -19.92], sales: 45 },
    { name: 'Curitiba', coordinates: [-49.27, -25.43], sales: 38 },
  ];

  // `mapColors()` reads the theme, so this re-runs when the scheme or preset changes.
  const layers = $derived.by(() => {
    const c = mapColors();
    return [
      new ScatterplotLayer({
        id: 'stores',
        data: stores,
        getPosition: (d: (typeof stores)[number]) => d.coordinates as [number, number],
        getRadius: (d: (typeof stores)[number]) => d.sales * 600,
        getFillColor: c.brandMuted,
        getLineColor: c.brand,
        stroked: true,
        lineWidthMinPixels: 2,
      }),
    ];
  });
</script>

<div class="stage">
  <MapStage
    {styles}
    {layers}
    label="Store map"
    fit={[[-50, -26], [-42, -19]]}
    padding={{ top: 24, right: 24, bottom: 24, left: 24 }}
  >
    <MapControls />
  </MapStage>
</div>

<style>
  .stage { width: 100%; height: 360px; border-radius: var(--ca-radius-lg); overflow: hidden; }
</style>
