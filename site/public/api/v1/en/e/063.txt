Element: #063 Submit state (inputs/use-form)
What it shows: Form with a schema that simulates a 1.2 second submit, then shows a success message and resets.
When to use: Create or edit forms that show a loading state while saving.
Spec:
- `useForm(` with `schema: (data) => {` requires a non-empty title
- `submit()` calls `form.validate()`, sets `form.submitting = true`, waits `setTimeout(r, 1200)`, then `form.reset()`
- TextField Title and TextArea Notes use `disabled={form.submitting}`
- Submit button uses `loading={form.submitting}`, label is `{form.submitting ? 'Saving…' : 'Submit'}`
- Success text renders in `<p class="hint hint--success" role="status">`
- Reset button has `variant="outlined"` and clears `result`
Tags: use-form, submit, loading, schema, validation, async
Needs: @codeatlasdev/ui@4.1.0
Components: Button, TextArea, TextField
Live demo: https://ui.codeatlas.com.br/docs/inputs/use-form#submit-state

<script lang="ts">
  import { Button, TextArea, TextField, useForm } from '@codeatlasdev/ui';

  const form = useForm(
    { title: '', notes: '' },
    {
      schema: (data) => {
        const errs: Partial<Record<'title' | 'notes', string>> = {};
        if (!data.title.trim()) errs.title = 'Title is required.';
        return errs;
      },
    },
  );

  let result = $state('');

  async function submit() {
    if (!form.validate()) return;
    form.submitting = true;
    result = '';
    await new Promise((r) => setTimeout(r, 1200));
    form.submitting = false;
    result = `Submitted: "${form.fields.title.value}"`;
    form.reset();
  }
</script>

<div class="col">
  <TextField
    label="Title"
    placeholder="My item title"
    value={form.fields.title.value}
    error={form.fields.title.error}
    oninput={(v) => form.set('title', v)}
    onblur={() => form.touch('title')}
    disabled={form.submitting}
  />
  <TextArea
    label="Notes"
    placeholder="Optional notes…"
    value={form.fields.notes.value}
    error={form.fields.notes.error}
    oninput={(v) => form.set('notes', v)}
    onblur={() => form.touch('notes')}
    disabled={form.submitting}
  />
  <div class="row">
    <Button onclick={submit} loading={form.submitting}>
      {form.submitting ? 'Saving…' : 'Submit'}
    </Button>
    <Button
      variant="outlined"
      onclick={() => {
        form.reset();
        result = '';
      }}
    >
      Reset
    </Button>
  </div>
  {#if result}
    <p class="hint hint--success" role="status">{result}</p>
  {/if}
</div>

  }
