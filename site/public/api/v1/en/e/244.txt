Element: #244 Native prompt (feedback/install-prompt)
What it shows: The install prompt in native mode, triggered by a stand-in beforeinstallprompt event fired on mount.
When to use: Offering an install banner on browsers that expose the beforeinstallprompt event, such as Chrome and Edge.
Spec:
- `<InstallPrompt mode="native" storageKey="demo-install-native" />`
- `onMount` dispatches `new Event('beforeinstallprompt', { cancelable: true })` on the window
- The fake event gets `prompt: () => Promise.resolve()` and `userChoice: Promise.resolve({ outcome: 'dismissed' })`
- Wrapped in `<div class="stage">` with `min-height: calc(var(--ca-space-8) * 3)`
Tags: install-prompt, pwa, native, beforeinstallprompt, banner
Needs: @codeatlasdev/ui@4.1.0
Components: InstallPrompt
Live demo: https://ui.codeatlas.com.br/docs/feedback/install-prompt#native-prompt

<script lang="ts">
  import { InstallPrompt } from '@codeatlasdev/ui';
  import { onMount } from 'svelte';

  // A real page waits for the browser's `beforeinstallprompt`. This demo fires a stand-in so the
  // banner can be seen anywhere; Install then resolves without opening a browser dialog.
  onMount(() => {
    const event = new Event('beforeinstallprompt', { cancelable: true });
    Object.assign(event, {
      prompt: () => Promise.resolve(),
      userChoice: Promise.resolve({ outcome: 'dismissed' }),
    });
    dispatchEvent(event);
  });
</script>

<!-- The prompt renders nothing until it is mounted in the browser; the box gives the island
     something to intersect (client:visible) and keeps the page from shifting when it appears. -->
<div class="stage">
  <InstallPrompt mode="native" storageKey="demo-install-native" />
</div>

<style>
  .stage { min-height: calc(var(--ca-space-8) * 3); }
</style>
