Element: #025 Phone and CEP (inputs/masked-field)
What it shows: A PhoneField and a CepField that looks up an address asynchronously and prints the resolved street, district and city.
When to use: Brazilian address and contact forms that autofill street data from a postal code lookup.
Spec:
- `<PhoneField label="WhatsApp" bind:value={phone} />`
- `<CepField` with `label="CEP"` and `description="Try 01310-100."`
- `bind:lookupPending={pending}`, `onlookup={lookup}` and `onresolve={(a) => (address = a)}`
- `async function lookup(raw: string)` waits 600 ms and returns a `CepAddress` only for `01310100`, otherwise `null`
- Result line shows `Looking up…` while pending, then `logradouro`, `bairro`, `localidade` and `uf`
Tags: phone-field, cep-field, lookup, address, brazil, async, forms
Needs: @codeatlasdev/ui@4.1.0
Components: CepField, PhoneField
Live demo: https://ui.codeatlas.com.br/docs/inputs/masked-field#phone-and-cep

<script lang="ts">
  import { CepField, PhoneField, type CepAddress } from '@codeatlasdev/ui/forms';

  let phone = $state('');
  let cep = $state('');
  let pending = $state(false);
  let address = $state<CepAddress | null>(null);

  // Demo lookup. Real apps inject ViaCEP (or any provider) here.
  async function lookup(raw: string): Promise<CepAddress | null> {
    await new Promise((r) => setTimeout(r, 600));
    return raw === '01310100'
      ? { logradouro: 'Avenida Paulista', bairro: 'Bela Vista', localidade: 'São Paulo', uf: 'SP' }
      : null;
  }
</script>

<div class="col">
  <PhoneField label="WhatsApp" bind:value={phone} />
  <p class="out">value = {phone || '–'}</p>
  <CepField
    label="CEP"
    description="Try 01310-100."
    bind:value={cep}
    bind:lookupPending={pending}
    onlookup={lookup}
    onresolve={(a) => (address = a)}
  />
  <p class="out">
    {#if pending}Looking up…{:else if address}{address.logradouro}, {address.bairro} · {address.localidade}/{address.uf}{:else}–{/if}
  </p>
</div>
