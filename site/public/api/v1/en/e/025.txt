Element: #025 Phone and CEP (inputs/masked-field)
What it shows: PhoneField binds E.164 (+55...) while showing the masked form. CepField resolves an address through onlookup. Try 01310-100.
Tags: phone, and, cep, cepfield, phonefield
Needs: @codeatlasdev/ui@4.1.0
Components: CepField, PhoneField
Live demo: https://ui.codeatlas.com.br/docs/inputs/masked-field#phone-and-cep

<script lang="ts">
  import { CepField, PhoneField, type CepAddress } from '@codeatlasdev/ui/forms';

  let phone = $state('');
  let cep = $state('');
  let pending = $state(false);
  let address = $state<CepAddress | null>(null);

  // Demo lookup. Real apps inject ViaCEP (or any provider) here.
  async function lookup(raw: string): Promise<CepAddress | null> {
    await new Promise((r) => setTimeout(r, 600));
    return raw === '01310100'
      ? { logradouro: 'Avenida Paulista', bairro: 'Bela Vista', localidade: 'São Paulo', uf: 'SP' }
      : null;
  }
</script>

<div class="col">
  <PhoneField label="WhatsApp" bind:value={phone} />
  <p class="out">value = {phone || '–'}</p>
  <CepField
    label="CEP"
    description="Try 01310-100."
    bind:value={cep}
    bind:lookupPending={pending}
    onlookup={lookup}
    onresolve={(a) => (address = a)}
  />
  <p class="out">
    {#if pending}Looking up…{:else if address}{address.logradouro}, {address.bairro} · {address.localidade}/{address.uf}{:else}–{/if}
  </p>
</div>
