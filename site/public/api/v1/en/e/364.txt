Element: #364 Stacked toasts (motion/transitions)
What it shows: Toasts that slide in from above, stack with a depth cascade (each older one lower, smaller and dimmer) and leave when dismissed.
When to use: In-app notifications that can arrive in bursts: payments, comments, exports, limit warnings.
Spec:
- `MotionStage` with `label="Stacked toasts: they enter, stack up and dismiss"` and `loop={1400}`
- At most `const MAX = 3` toasts; a new one goes first and the oldest is cut
- Depth comes from `--i`: `transform: translateY(calc(var(--i) * 0.9rem)) scale(calc(1 - var(--i) * 0.05))` and `opacity: calc(1 - var(--i) * 0.28)`
- Enter: `animation: ts-in 320ms cubic-bezier(0.2, 0, 0, 1)` from `translateY(-1.5rem) scale(0.96)`; restack: `transition: transform 320ms cubic-bezier(0.2, 0, 0, 1), opacity 320ms cubic-bezier(0.2, 0, 0, 1)`
- Only the front toast is announced and readable: older ones get `class:ts__toast--behind={i > 0}` (content `opacity: 0`), `aria-hidden` and `inert`, so the stack never overlaps text
- Dismiss: `out:leave` runs `duration: reduced ? 0 : 240` with `easing: cubicOut`, fading to `opacity:${t}` while `translate:${u * 2.5}rem 0` slides it aside
- Under reduced motion the stack is shown full with `animation: none`
Motion:
- opacity, transform: translateY and scale (enter): 320ms cubic-bezier(0.2, 0, 0, 1)
- opacity and translate (dismiss): 240 cubicOut
- transform and opacity (restack as the list changes): 320ms cubic-bezier(0.2, 0, 0, 1)

<script lang="ts">
  import { Button, MotionStage, type MotionCtx } from '@codeatlasdev/ui';
  import { CheckCircle, Info, TriangleAlert, X } from '@lucide/svelte';
  import { cubicOut } from 'svelte/easing';

  type Kind = 'success' | 'info' | 'warning';
  type Toast = { id: number; kind: Kind; title: string; body: string };
  const script: Omit<Toast, 'id'>[] = [
    { kind: 'success', title: 'Invoice paid', body: 'R$ 4.820 from Northwind' },
    { kind: 'info', title: 'New comment', body: 'Ana on "Q4 forecast"' },
    { kind: 'warning', title: 'Seat limit', body: '11 of 12 seats used' },
    { kind: 'success', title: 'Export ready', body: 'customers.csv, 2.4 MB' },
  ];
  const icons = { success: CheckCircle, info: Info, warning: TriangleAlert };
  const MAX = 3;

  let toasts = $state<Toast[]>([]);
  let n = 0;
  let stage = $state<MotionCtx>();
  let lastCycle = -1;

  function push() {
    const t = { ...script[n % script.length], id: n++ };
    toasts = [t, ...toasts].slice(0, MAX);
  }
  // Dismiss: slide up-and-out sideways while fading; absolute positioning keeps the rest from jumping.
  function leave(_: Element, { reduced }: { reduced: boolean }) {
    return {
      duration: reduced ? 0 : 240,
      easing: cubicOut,
      css: (t: number, u: number) => `opacity:${t}; translate:${u * 2.5}rem 0`,
    };
  }
  const dismiss = (id: number) => (toasts = toasts.filter((t) => t.id !== id));

  $effect(() => {
    if (!stage) return;
    if (stage.reduced) {
      if (!toasts.length) for (let i = 0; i < MAX; i++) push();
      return;
    }
    if (!stage.playing || stage.cycle === lastCycle) return;
    lastCycle = stage.cycle;
    // Two pushes, then the oldest leaves: enter, stack, dismiss.
    if (toasts.length >= MAX) dismiss(toasts[toasts.length - 1].id);
    else push();
  });
</script>

<MotionStage label="Stacked toasts: they enter, stack up and dismiss" loop={1400}>
  {#snippet children(m)}
    <div class="ts" {@attach () => { stage = m; }}>
      <div class="ts__bar">
        <Button size="sm" onclick={push}>Show toast</Button>
      </div>
      <ol class="ts__stack" aria-label="Notifications">
        {#each toasts as t, i (t.id)}
          {@const Icon = icons[t.kind]}
          <li class="ts__item" class:ts__item--instant={m.reduced} style="--i:{i}" out:leave={{ reduced: m.reduced }}>
            <div class="ts__toast ts__toast--{t.kind}" class:ts__toast--behind={i > 0} aria-hidden={i > 0 ? 'true' : undefined} inert={i > 0}>
              <Icon size={18} aria-hidden="true" />
              <div class="ts__text">
                <p class="ts__title">{t.title}</p>
                <p class="ts__body">{t.body}</p>
              </div>
              <button type="button" class="ts__x" aria-label={`Dismiss ${t.title}`} onclick={() => dismiss(t.id)}><X size={14} aria-hidden="true" /></button>
            </div>
          </li>
        {/each}
      </ol>
    </div>
  {/snippet}
</MotionStage>
