Element: #239 Leading icons and brand logos (navigation/menu)
What it shows: A dropdown menu of agents where brand-colored icons opt out of the item's text color, including a scheme-aware monochrome mark.
When to use: Pickers that list third-party products or agents with their real logos, such as a new-tab or integration menu.
Spec:
- `<DropdownMenu>` with `{#snippet triggerChild(props)}` rendering `<Button variant="glass" {...props}>New tab</Button>`
- `<DropdownMenuLabel>Agents</DropdownMenuLabel>` followed by 4 items: Shell, Claude Code, Codex, Coming soon
- Shell uses `{#snippet icon()}<Bot />{/snippet}`; Claude Code and Codex set `brandIcon` so the inline `svg` keeps its own colors
- Codex wraps `<SchemeAsset>` with `{#snippet light()}` and `{#snippet dark()}` svg variants
- Last item is `brandIcon disabled`
- Each item sets `last` through `onclick`, shown as `Opened: {last}` with `aria-live="polite"`
Tags: dropdown-menu, menu, brand-icon, scheme-asset, dark-mode, icon, disabled
Needs: @codeatlasdev/ui@4.1.0
Components: DropdownMenu, DropdownMenuItem, DropdownMenuLabel, SchemeAsset, Button
Live demo: https://ui.codeatlas.com.br/docs/navigation/menu#leading-icons-and-brand-logos

<script lang="ts">
  import { DropdownMenu, DropdownMenuItem, DropdownMenuLabel, SchemeAsset, Button } from '@codeatlasdev/ui';
  import { Bot } from '@lucide/svelte';

  let last = $state('');
</script>

<DropdownMenu>
  {#snippet triggerChild(props)}
    <Button variant="glass" {...props}>New tab</Button>
  {/snippet}
  <DropdownMenuLabel>Agents</DropdownMenuLabel>
  <DropdownMenuItem onclick={() => (last = 'Shell')}>
    {#snippet icon()}<Bot />{/snippet}
    Shell
  </DropdownMenuItem>
  <!-- A brand mark keeps its own colors: brandIcon opts it out of the item's text color. -->
  <DropdownMenuItem brandIcon onclick={() => (last = 'Claude Code')}>
    {#snippet icon()}
      <svg viewBox="0 0 16 16"><rect width="16" height="16" rx="4" fill="#d97757" /><path d="M5 11 8 4l3 7" fill="none" stroke="#fff" stroke-width="1.6" /></svg>
    {/snippet}
    Claude Code
  </DropdownMenuItem>
  <!-- A monochrome mark needs a black asset on light and a white one on dark: SchemeAsset picks by the kit's scheme. -->
  <DropdownMenuItem brandIcon onclick={() => (last = 'Codex')}>
    {#snippet icon()}
      <SchemeAsset>
        {#snippet light()}<svg viewBox="0 0 16 16"><circle cx="8" cy="8" r="7" fill="#111" /></svg>{/snippet}
        {#snippet dark()}<svg viewBox="0 0 16 16"><circle cx="8" cy="8" r="7" fill="#fff" /></svg>{/snippet}
      </SchemeAsset>
    {/snippet}
    Codex
  </DropdownMenuItem>
  <DropdownMenuItem brandIcon disabled>
    {#snippet icon()}
      <svg viewBox="0 0 16 16"><rect width="16" height="16" rx="4" fill="#d97757" /></svg>
    {/snippet}
    Coming soon
  </DropdownMenuItem>
</DropdownMenu>

{#if last}<span class="note" aria-live="polite">Opened: {last}</span>{/if}

<style>
  .note { font-size: var(--ca-font-size-body2); color: var(--ca-text-secondary); }
</style>
