Element: #204 Real-time stream (charts/line)
What it shows: A live line chart that shifts in a new Live revenue point every 800 ms over a 20-point window.
When to use: Real-time monitoring such as live revenue, traffic or latency streams.
Spec:
- `let streamData = $state<number[]>` starts with 20 values from `68000` to `115000`
- `setInterval` in `onMount` runs every `800` ms and sets `streamData = [...streamData.slice(1), next]`
- `next` is clamped with `Math.min(125000, Math.max(42000,`
- `series={[{ name: 'Live revenue', data: streamData }]}`
- `categories={streamData.map((_, i) => String(i))}`
- `valueFormatter={fmt}` and `height="300px"`
Tags: line-chart, streaming, live, realtime, onmount, interval
Needs: @codeatlasdev/ui@4.1.0 + echarts
Components: Card, LineChart
Live demo: https://ui.codeatlas.com.br/docs/charts/line#real-time-stream

<script lang="ts">
  import { Card } from '@codeatlasdev/ui';
  import { LineChart } from '@codeatlasdev/ui/charts';
  import { onMount } from 'svelte';

  const fmt = (v: number) => `$${(v / 1000).toFixed(0)}k`;

  let streamData = $state<number[]>([
    68000, 71000, 65000, 73000, 79000, 76000, 82000, 85000, 78000, 88000,
    91000, 86000, 94000, 98000, 103000, 99000, 107000, 112000, 108000, 115000,
  ]);

  onMount(() => {
    const id = setInterval(() => {
      const last = streamData[streamData.length - 1];
      const next = Math.min(125000, Math.max(42000, last + (Math.random() - 0.45) * 8000));
      streamData = [...streamData.slice(1), next];
    }, 800);
    return () => clearInterval(id);
  });
</script>

<Card material="glass" full>
  <LineChart
    series={[{ name: 'Live revenue', data: streamData }]}
    categories={streamData.map((_, i) => String(i))}
    valueFormatter={fmt}
    height="300px"
  />
</Card>
