Element: #139 Default (scroll) (display/table)
What it shows: A default SimpleTable listing three users with name, email, role and status in a flush glass card.
When to use: Compact read-only tables that keep their columns and scroll horizontally when space runs out.
Spec:
- `<Card material="glass" flush full>` wraps the table
- `<SimpleTable>` with no props, plain `thead` and `tbody`
- Header columns: Name, Email, Role, Status
- 3 rows from `rows` with `name`, `email`, `role`, `status`
- The first cell uses `class="primary"`
Tags: table, simple-table, scroll, overflow, card, users
Needs: @codeatlasdev/ui@4.1.0
Components: SimpleTable, Card
Live demo: https://ui.codeatlas.com.br/docs/display/table#default-scroll

<script lang="ts">
  import { SimpleTable, Card } from '@codeatlasdev/ui';

  const rows = [
    { name: 'Ana Souza', email: 'ana@example.com', role: 'Admin', status: 'Active' },
    { name: 'Bruno Lima', email: 'bruno@example.com', role: 'Editor', status: 'Invited' },
    { name: 'Carla Dias', email: 'carla@example.com', role: 'Viewer', status: 'Active' },
  ];
</script>

<Card material="glass" flush full>
  <SimpleTable>
    <thead>
      <tr><th>Name</th><th>Email</th><th>Role</th><th>Status</th></tr>
    </thead>
    <tbody>
      {#each rows as r}
        <tr>
          <td class="primary">{r.name}</td>
          <td>{r.email}</td>
          <td>{r.role}</td>
          <td>{r.status}</td>
        </tr>
      {/each}
    </tbody>
  </SimpleTable>
</Card>
