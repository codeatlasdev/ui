Element: #081 Basic and destructive (feedback/alert-dialog)
What it shows: Two alert dialogs, a plain sign out confirmation and a destructive project delete, with an event log below.
When to use: Asking the user to confirm an interrupting action such as signing out or deleting a project.
Spec:
- Sign out dialog: `bind:open={signOutOpen}`, `confirmLabel="Sign out"`, `onconfirm` and `oncancel` handlers
- Delete dialog: `bind:open={deleteOpen}`, `confirmLabel="Delete"`, `destructive` flag, only `onconfirm`
- Triggers are `<Button variant="outlined"` and `<Button variant="destructive"` with `{#snippet leading()}` icons `<LogOut size={14} />` and `<Trash2 size={14} />`
- Handlers call `push('Signed out')`, `push('Sign out cancelled')` and `push('Project deleted')`
- Log keeps the latest four entries with `slice(0, 4)` in a `<ul class="log" aria-live="polite">`
Tags: alert-dialog, confirm, destructive, sign-out, delete, dialog
Needs: @codeatlasdev/ui@4.1.0
Components: AlertDialog, Button
Live demo: https://ui.codeatlas.com.br/docs/feedback/alert-dialog#basic-and-destructive

<script lang="ts">
  import { AlertDialog, Button } from '@codeatlasdev/ui';
  import { Trash2, LogOut } from '@lucide/svelte';

  let signOutOpen = $state(false);
  let deleteOpen = $state(false);
  let log = $state<string[]>([]);
  const push = (m: string) => (log = [m, ...log].slice(0, 4));
</script>

<Button variant="outlined" onclick={() => (signOutOpen = true)}>
  {#snippet leading()}<LogOut size={14} />{/snippet}
  Sign out
</Button>
<Button variant="destructive" onclick={() => (deleteOpen = true)}>
  {#snippet leading()}<Trash2 size={14} />{/snippet}
  Delete project
</Button>

<AlertDialog
  bind:open={signOutOpen}
  title="Sign out?"
  description="You will need to sign in again to access your workspace."
  confirmLabel="Sign out"
  onconfirm={() => push('Signed out')}
  oncancel={() => push('Sign out cancelled')}
/>

<AlertDialog
  bind:open={deleteOpen}
  title="Delete this project?"
  description="This action cannot be undone. All data will be permanently removed."
  confirmLabel="Delete"
  destructive
  onconfirm={() => push('Project deleted')}
/>

{#if log.length}
  <ul class="log" aria-live="polite">
    {#each log as entry, i (i + entry)}<li>{entry}</li>{/each}
  </ul>
{/if}

<style>
  .log {
    flex-basis: 100%;
    list-style: none;
    margin: 0;
    padding: 0;
    display: grid;
    gap: 4px;
    font-size: 13px;
    color: var(--ca-text-secondary);
  }
</style>
