Element: #345 Copy to check (motion/micro)
What it shows: A copy button whose two sheets scale away while a check mark draws itself, then it reverts after 1.6 seconds.
When to use: Copy-to-clipboard buttons next to a command, token or link, where the icon confirms the action.
Spec:
- `MotionStage` from `@codeatlasdev/ui` with `loop={2800}`
- `navigator.clipboard?.writeText(CODE)` on a real click; autoplay never touches the clipboard
- `.cc__btn--done .cc__sheets` goes to `transform: scale(0.6)` and `opacity: 0`
- `stroke-dasharray: 1;` with `stroke-dashoffset: 1` drawing to `stroke-dashoffset: 0` on the tick path
- `setTimeout(() => (copied = false), 1600)` reverts the icon
- `announce('Copied to clipboard')` and a `ca-sr-only` label carry the state for screen readers
Motion:
- transform, opacity (sheets): 280ms cubic-bezier(0.2, 0, 0, 1)
- stroke-dashoffset (check tick): 320ms cubic-bezier(0.2, 0, 0, 1) delay 120ms
Names: Copy to check | Copy confirmation | Copiar e confirmar | Copier puis valider
Tags: motion, copy, check, morph, icon, clipboard
Needs: @codeatlasdev/ui@4.3.0
Components: MotionStage
Live demo: https://ui.codeatlas.com.br/docs/motion/micro#copy-to-check

<script lang="ts">
  import { MotionStage, announce } from '@codeatlasdev/ui';

  const CODE = 'npm i @codeatlasdev/ui';
  let copied = $state(false);
  let timer: ReturnType<typeof setTimeout> | undefined;

  function copy(real: boolean) {
    if (real) navigator.clipboard?.writeText(CODE).catch(() => {});
    copied = true;
    announce('Copied to clipboard');
    clearTimeout(timer);
    timer = setTimeout(() => (copied = false), 1600);
  }

  // Autoplay "clicks" copy without touching the real clipboard.
  const autoplay = (node: HTMLElement) => {
    copied = false;
    const t = setTimeout(() => copy(false), 500);
    return () => {
      clearTimeout(t);
      clearTimeout(timer);
      copied = false;
    };
  };
</script>

<MotionStage label="Copy to check: the copy icon morphs into a check mark" loop={2800}>
  {#snippet children(m)}
    {#key m.cycle}
      <div class="cc" {@attach (node) => (m.playing && !m.reduced ? autoplay(node) : undefined)}>
        <code class="cc__code">{CODE}</code>
        <button
          type="button"
          class="cc__btn"
          class:cc__btn--done={copied}
          class:cc__btn--still={m.reduced}
          onclick={() => copy(true)}
        >
          <svg viewBox="0 0 24 24" class="cc__icon" aria-hidden="true">
            <!-- Copy: two sheets. Check: one stroke that draws itself. They cross-fade and the sheets slide into the tick. -->
            <g class="cc__sheets">
              <rect x="9" y="9" width="11" height="11" rx="2" />
              <path d="M5 15V6a2 2 0 0 1 2-2h9" />
            </g>
            <path class="cc__tick" d="M5 12.5l4.5 4.5L19 7.5" pathLength="1" />
          </svg>
          <span class="ca-sr-only">{copied ? 'Copied' : 'Copy command'}</span>
        </button>
      </div>
    {/key}
  {/snippet}
</MotionStage>
