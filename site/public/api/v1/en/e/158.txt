Element: #158 Default (navigation/menu)
What it shows: A dropdown menu of file actions with icons, a separator and a destructive Delete item, reporting the last action.
When to use: Row or toolbar action menus offering several operations, with the dangerous one set apart.
Spec:
- `<DropdownMenu>` with `{#snippet triggerChild(props)}` rendering `<Button variant="glass" {...props}>Open menu</Button>`
- `<DropdownMenuLabel>File actions</DropdownMenuLabel>` then items Edit, Duplicate, Download
- Each item has `{#snippet icon()}` with `<Edit2 />`, `<Copy />`, `<Download />` or `<Trash2 />`
- `<DropdownMenuSeparator />` precedes `<DropdownMenuItem variant="destructive"` for Delete
- `let last = $state('')` is set per item and shown as `Last action: {last}` with `aria-live="polite"`
Tags: dropdown-menu, menu, actions, destructive, separator, icon, trigger
Needs: @codeatlasdev/ui@4.1.0
Components: DropdownMenu, DropdownMenuItem, DropdownMenuSeparator, DropdownMenuLabel, Button
Live demo: https://ui.codeatlas.com.br/docs/navigation/menu#default

<script lang="ts">
  import { DropdownMenu, DropdownMenuItem, DropdownMenuSeparator, DropdownMenuLabel, Button } from '@codeatlasdev/ui';
  import { Edit2, Copy, Trash2, Download } from '@lucide/svelte';

  let last = $state('');
</script>

<DropdownMenu>
  {#snippet triggerChild(props)}
    <Button variant="glass" {...props}>Open menu</Button>
  {/snippet}
  <DropdownMenuLabel>File actions</DropdownMenuLabel>
  <DropdownMenuItem onclick={() => (last = 'Edit')}>
    {#snippet icon()}<Edit2 />{/snippet}
    Edit
  </DropdownMenuItem>
  <DropdownMenuItem onclick={() => (last = 'Duplicate')}>
    {#snippet icon()}<Copy />{/snippet}
    Duplicate
  </DropdownMenuItem>
  <DropdownMenuItem onclick={() => (last = 'Download')}>
    {#snippet icon()}<Download />{/snippet}
    Download
  </DropdownMenuItem>
  <DropdownMenuSeparator />
  <DropdownMenuItem variant="destructive" onclick={() => (last = 'Delete')}>
    {#snippet icon()}<Trash2 />{/snippet}
    Delete
  </DropdownMenuItem>
</DropdownMenu>

{#if last}<span class="note" aria-live="polite">Last action: {last}</span>{/if}

<style>
  .note { font-size: var(--ca-font-size-body2); color: var(--ca-text-secondary); }
</style>
