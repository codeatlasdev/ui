Element: #090 Left and bottom (feedback/drawer)
What it shows: Two drawers opened from separate buttons: a 280px left navigation drawer and a bottom sort drawer.
When to use: Mobile style navigation from the left and option pickers sliding up from the bottom.
Spec:
- Left: `<Drawer bind:open={leftOpen} title="Navigation" side="left" width="280px">` with 5 items: Dashboard, Users, Products, Reports, Settings
- Bottom: `<Drawer bind:open={bottomOpen} title="Sort by" side="bottom">` with 5 options: Name (A-Z), Name (Z-A), Date joined, Last active, Role
- Items are native `<button class="item"` elements inside `<Stack gap="1">` generated by `{#each`
- Each item closes its drawer with `onclick={() => (leftOpen = false)}` or `onclick={() => (bottomOpen = false)}`
- Triggers are two `<Button variant="outlined"` labelled Open left and Open bottom
Tags: drawer, left, bottom, navigation, sort, side-panel
Needs: @codeatlasdev/ui@4.1.0
Components: Drawer, Button, Stack
Live demo: https://ui.codeatlas.com.br/docs/feedback/drawer#left-and-bottom

<script lang="ts">
  import { Drawer, Button, Stack } from '@codeatlasdev/ui';

  let leftOpen = $state(false);
  let bottomOpen = $state(false);
</script>

<Button variant="outlined" onclick={() => (leftOpen = true)}>Open left</Button>
<Button variant="outlined" onclick={() => (bottomOpen = true)}>Open bottom</Button>

<Drawer bind:open={leftOpen} title="Navigation" side="left" width="280px">
  <Stack gap="1">
    {#each ['Dashboard', 'Users', 'Products', 'Reports', 'Settings'] as item}
      <button class="item" onclick={() => (leftOpen = false)}>{item}</button>
    {/each}
  </Stack>
</Drawer>

<Drawer bind:open={bottomOpen} title="Sort by" side="bottom">
  <Stack gap="1">
    {#each ['Name (A-Z)', 'Name (Z-A)', 'Date joined', 'Last active', 'Role'] as opt}
      <button class="item" onclick={() => (bottomOpen = false)}>{opt}</button>
    {/each}
  </Stack>
</Drawer>
