Element: #166 Live shell (layout/app-shell)
What it shows: A working app shell with a rail, panel and page content whose pathname lives in local state and updates on navigation.
When to use: Wiring the shell into an app: the router owns the URL and the shell receives it as a prop.
Spec:
- `<AppShellLayout {modules} {pathname} onNavigate={(path) => (pathname = path)} persistPanel={false} mainId={null} skipLink={false}>`
- `const modules: NavModule[]` has two modules, `id: 'app'` and `id: 'account'`, with 3 items each
- `let pathname = $state('/app/dashboard')` is the only navigation state
- `function intercept(e: MouseEvent)` turns in-stage `a[href^="/"]` clicks into `pathname` updates
- Page is `<PageShell title={titles[pathname] ?? 'Page'} description={pathname}>` with a `KpiGrid` of 2 `KpiCard` items: Revenue and Users
- Stage height is `height: 440px`
Tags: app-shell, app-shell-layout, navigation, rail, sidebar, router-agnostic, page-shell
Needs: @codeatlasdev/ui@4.1.0
Components: AppShellLayout, PageShell, KpiCard, KpiGrid, Card
Live demo: https://ui.codeatlas.com.br/docs/layout/app-shell#live-shell

<script lang="ts">
  import { AppShellLayout, PageShell, KpiCard, KpiGrid, Card, type NavModule } from '@codeatlasdev/ui';
  import { LayoutDashboard, Users, BarChart3, Settings, CreditCard, Activity } from '@lucide/svelte';

  const modules: NavModule[] = [
    {
      id: 'app',
      label: 'App',
      icon: LayoutDashboard,
      basePath: '/app',
      items: [
        { id: 'dashboard', label: 'Dashboard', icon: LayoutDashboard, path: '/app/dashboard' },
        { id: 'users', label: 'Users', icon: Users, path: '/app/users' },
        { id: 'reports', label: 'Reports', icon: BarChart3, path: '/app/reports' },
      ],
    },
    {
      id: 'account',
      label: 'Account',
      icon: Settings,
      basePath: '/account',
      items: [
        { id: 'billing', label: 'Billing', icon: CreditCard, path: '/account/billing' },
        { id: 'activity', label: 'Activity', icon: Activity, path: '/account/activity' },
        { id: 'settings', label: 'Settings', icon: Settings, path: '/account/settings' },
      ],
    },
  ];

  // The shell is router-agnostic: the app owns the URL and feeds it back in.
  let pathname = $state('/app/dashboard');

  // Panel items are plain <a href> links, exactly what a client router intercepts. With no router
  // in this demo, do the same: take over in-stage link clicks and feed the path back in.
  function intercept(e: MouseEvent) {
    if (e.defaultPrevented || e.button !== 0 || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey) return;
    const a = (e.target as Element).closest('a[href^="/"]');
    if (!a) return;
    e.preventDefault();
    pathname = a.getAttribute('href')!;
  }

  const titles: Record<string, string> = {
    '/app/dashboard': 'Dashboard',
    '/app/users': 'Users',
    '/app/reports': 'Reports',
    '/account/billing': 'Billing',
    '/account/activity': 'Activity',
    '/account/settings': 'Settings',
  };
</script>

<!-- svelte-ignore a11y_no_static_element_interactions, a11y_click_events_have_key_events -->
<div class="stage" onclick={intercept}>
  <AppShellLayout {modules} {pathname} onNavigate={(path) => (pathname = path)} persistPanel={false} mainId={null} skipLink={false}>
    <PageShell title={titles[pathname] ?? 'Page'} description={pathname}>
      <KpiGrid>
        <KpiCard title="Revenue" value="$118k" trend={23.4} />
        <KpiCard title="Users" value="24.8k" trend={8.1} />
      </KpiGrid>
      <Card material="glass" full>
        <p class="hint">Click a rail icon or a panel item. The demo keeps the pathname in local state.</p>
      </Card>
    </PageShell>
  </AppShellLayout>
</div>
