Element: #481 Color, count, actions and reorder (layout/app-shell)
What it shows: App shell panel rows with a color dot, a neutral count, a per-row actions menu and drag-to-reorder.
When to use: Lists of user-owned things (workspaces, folders, saved views) that need a marker, a total, quick actions and a custom order.
Spec:
- `<AppShellLayout {modules} {pathname} reorder={NavReorder} itemMenu={NavItemMenu} persistPanel={false} mainId={null} skipLink={false}>`: reorder and the item menu are opt-in imports
- One module `id: 'projects'` with a non-collapsible section `label: 'Workspaces'` and `onReorder: reorder`
- 3 items: Design system (#e11d48, count 42), Marketing site (#2563eb, count 1280, shown as 1.2k), Mobile app (#16a34a, count 0)
- Every item has `color`, `count` and `actions`; `actions` is a snippet with 2 `DropdownMenuItem`s: Favorite and Archive (destructive)
- `reorder(e: NavReorderEvent)` uses `next.splice(e.from, 1)` and `next.splice(e.to, 0, moved)`, then writes the visible `log` line; the drag itself is announced by the shell
- A second module `id: 'other'` with `items: []`
- Stage height is `height: 360px`
Tags: app-shell, nav-item, color, count, actions-menu, reorder, drag-and-drop, sidebar
Needs: @codeatlasdev/ui@4.3.0
Components: AppShellLayout, DropdownMenuItem, NavItemMenu, NavReorder, PageShell, Card
Live demo: https://ui.codeatlas.com.br/docs/layout/app-shell#color-count-actions-and-reorder

<script lang="ts">
  import {
    AppShellLayout,
    DropdownMenuItem,
    NavItemMenu,
    NavReorder,
    PageShell,
    Card,
    type NavModule,
    type NavReorderEvent,
  } from '@codeatlasdev/ui';
  import { Folder } from '@lucide/svelte';

  type Row = { id: string; name: string; color: string; count: number };
  let rows = $state<Row[]>([
    { id: 'design', name: 'Design system', color: '#e11d48', count: 42 },
    { id: 'web', name: 'Marketing site', color: '#2563eb', count: 1280 },
    { id: 'mobile', name: 'Mobile app', color: '#16a34a', count: 0 },
  ]);
  let log = $state('Open the … menu of a row, or drag a row by its grip.');

  function reorder(e: NavReorderEvent) {
    const next = rows.slice();
    const [moved] = next.splice(e.from, 1);
    next.splice(e.to, 0, moved);
    rows = next;
    log = `Moved ${moved.name} to position ${e.to + 1}.`;
  }

  const modules: NavModule[] = $derived([
    {
      id: 'projects',
      label: 'Projects',
      icon: Folder,
      basePath: '/projects',
      items: [
        {
          label: 'Workspaces',
          collapsible: false,
          onReorder: reorder,
          items: rows.map((r) => ({
            id: r.id,
            label: r.name,
            icon: Folder,
            path: `/projects/${r.id}`,
            color: r.color,
            count: r.count,
            actions: menu,
          })),
        },
      ],
    },
    { id: 'other', label: 'Other', icon: Folder, basePath: '/other', items: [] },
  ]);

  let pathname = $state('/projects/design');
  function intercept(e: MouseEvent) {
    if (e.defaultPrevented || e.button !== 0 || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey) return;
    const a = (e.target as Element).closest('a[href^="/"]');
    if (!a) return;
    e.preventDefault();
    pathname = a.getAttribute('href')!;
  }
</script>

{#snippet menu()}
  <DropdownMenuItem onclick={() => (log = 'Favorite')}>{#snippet children()}Favorite{/snippet}</DropdownMenuItem>
  <DropdownMenuItem onclick={() => (log = 'Archive')} variant="destructive">{#snippet children()}Archive{/snippet}</DropdownMenuItem>
{/snippet}

<!-- svelte-ignore a11y_no_static_element_interactions, a11y_click_events_have_key_events -->
<div class="stage" onclick={intercept}>
  <AppShellLayout {modules} {pathname} reorder={NavReorder} itemMenu={NavItemMenu} persistPanel={false} mainId={null} skipLink={false}>
    <PageShell title="Workspaces" description={pathname}>
      <Card material="glass" full>
        <p class="hint">{log}</p>
      </Card>
    </PageShell>
  </AppShellLayout>
</div>

  }

  }
