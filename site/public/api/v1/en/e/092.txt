Element: #092 Animated (feedback/progress)
What it shows: Start a simulated upload.
Tags: animated, progress, button
Needs: @codeatlasdev/ui@4.0.0
Components: Progress, Button
Live demo: https://ui.codeatlas.com.br/docs/feedback/progress#animated

<script lang="ts">
  import { Progress, Button } from '@codeatlasdev/ui';
  import { onDestroy } from 'svelte';

  let value = $state(0);
  let running = $state(false);
  let timer: ReturnType<typeof setInterval> | undefined;

  function start() {
    if (running) return;
    value = 0;
    running = true;
    timer = setInterval(() => {
      value = Math.min(100, value + 5);
      if (value >= 100) {
        clearInterval(timer);
        running = false;
      }
    }, 100);
  }

  onDestroy(() => clearInterval(timer));
</script>

<div class="col">
  <div class="row">
    <span class="label">{value}%</span>
    <Progress {value} label="Upload progress" />
  </div>
  <Button variant="outlined" size="sm" disabled={running} onclick={start}>
    {running ? 'Uploading…' : 'Start upload'}
  </Button>
</div>
