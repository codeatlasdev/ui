Element: #324 Horizontal scroll section (motion/scroll)
What it shows: Vertical scrolling inside a pinned frame slides a row of four step cards sideways, with a progress bar underneath. The page scrolls down; the content moves across.
When to use: Onboarding steps, feature tours and case-study timelines where a sequence reads better left to right than stacked.
Spec:
- `MotionStage` from `@codeatlasdev/ui` with `loop={5600}`: autoplay scrolls the container, a wheel or touch scroll takes over
- `.hs__pin` is `position: sticky;` and keeps the frame fixed while `block-size: 30rem;` on `.hs__track` provides the scroll distance
- The rail is `inline-size: 400%` (four slides of `25%`) and travels `-75%` on X, so the last slide ends exactly in view
- Native: `animation-timeline: --hs` over `scroll-timeline: --hs block` on the scroller (the pin is `overflow: hidden`, so `scroll()` would bind to it); `@keyframes slide` runs from `translateX(0)` to `translateX(-75%)`, inside `@supports (animation-timeline: scroll())`
- Fallback without support: a `scroll` listener writes `--p` and the rail uses `translateX(calc(var(--p) * -75%))`
- Under reduced motion `--p` is `1`: the last slide is shown and the container does not scroll by itself
Motion:
- transform: translateX: animation-timeline: --hs linear
Names: Horizontal scroll section | Scroll-jacked horizontal | Rolagem horizontal por scroll vertical | Défilement horizontal

<script lang="ts">
  import { MotionStage, type MotionCtx } from '@codeatlasdev/ui';
  import type { Attachment } from 'svelte/attachments';

  const native = typeof CSS !== 'undefined' && CSS.supports('animation-timeline: scroll()');

  const slides = [
    { n: '01', title: 'Capture', body: 'Scan a receipt, the amount and merchant fill in.' },
    { n: '02', title: 'Categorize', body: 'Rules sort it into Travel, Meals or Software.' },
    { n: '03', title: 'Approve', body: 'Managers approve from the lock screen.' },
    { n: '04', title: 'Reimburse', body: 'Paid out on the next Friday run.' }
  ];

  // Fallback: write vertical progress (0..1) to --p; the track translates by -p * (slides - 1) * 100% of a slide.
  const track = (m: MotionCtx): Attachment<HTMLElement> => (el) => {
    if (native) return;
    const write = () => {
      const max = el.scrollHeight - el.clientHeight;
      if (!m.reduced) el.style.setProperty('--p', String(max > 0 ? el.scrollTop / max : 0));
    };
    write();
    el.addEventListener('scroll', write, { passive: true });
    return () => el.removeEventListener('scroll', write);
  };

  // Autoplay = scroll down in 3200ms (ease-in-out), which slides the track sideways; wheel or touch = takeover.
  const drive = (m: MotionCtx): Attachment<HTMLElement> => (el) => {
    const stop = () => m.takeover();
    el.addEventListener('wheel', stop, { passive: true });
    el.addEventListener('touchmove', stop, { passive: true });
    let raf = 0;
    if (m.playing) {
      void m.cycle;
      el.scrollTop = 0;
      const to = el.scrollHeight - el.clientHeight;
      const t0 = performance.now();
      const step = (now: number) => {
        if (!m.playing) return; // taken over or paused: never write a frame after the user starts scrolling
        const t = Math.min(1, (now - t0) / 3200);
        el.scrollTop = to * (t < 0.5 ? 4 * t * t * t : 1 - (-2 * t + 2) ** 3 / 2);
        if (t < 1) raf = requestAnimationFrame(step);
      };
      raf = requestAnimationFrame(step);
    }
    return () => {
      cancelAnimationFrame(raf);
      el.removeEventListener('wheel', stop);
      el.removeEventListener('touchmove', stop);
    };
  };
</script>

<MotionStage label="Horizontal scroll section: scrolling down slides four cards sideways" loop={5600}>
  {#snippet children(m)}
    <!-- svelte-ignore a11y_no_noninteractive_tabindex -->
    <div
      class="hs"
      class:hs--native={native}
      class:hs--final={m.reduced}
      role="region"
      aria-label="Expense flow, four steps"
      tabindex="0"
      {@attach track(m)}
      {@attach drive(m)}
    >
      <div class="hs__pin">
        <ol class="hs__rail">
          {#each slides as s (s.n)}
            <li class="hs__slide">
              <span class="hs__n">{s.n}</span>
              <p class="hs__title">{s.title}</p>
              <p class="hs__body">{s.body}</p>
            </li>
          {/each}
        </ol>
        <div class="hs__bar" aria-hidden="true"><span class="hs__fill"></span></div>
      </div>
      <div class="hs__track" aria-hidden="true"></div>
    </div>
  {/snippet}
</MotionStage>
