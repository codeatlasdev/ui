Element: #380 Cursor spotlight (motion/hero-ambient)
What it shows: A pricing grid lit by a soft brand spotlight and a grid of fine lines that only shows inside the light; the light eases toward the pointer.
When to use: Hero or feature grids where hovering should reveal structure: pricing tables, bento grids, docs landing pages.
Spec:
- `MotionStage` with `loop={1600}`; each cycle moves the light to the next entry of `waypoints`
- `@property --x` and `@property --y` are registered percentages, so `transition: --x 520ms cubic-bezier(0.2, 0, 0, 1)` eases them
- The pointer sets `--x` and `--y` from `getBoundingClientRect()`; `onpointerleave` hands control back to the autoplay waypoint
- One mask: `radial-gradient(circle 8rem at var(--x) var(--y), #000, transparent 72%)` reveals the line grid, `circle 10rem` paints the brand glow
- `@media (pointer: coarse)` sets `touch-action: none` so a finger drags the light instead of scrolling the demo
- Under reduced motion `class:spot--run={!m.reduced}` removes the transition and the light rests at `50%` / `40%`
Motion:
- --x, --y (radial mask center): 520ms cubic-bezier(0.2, 0, 0, 1)
Names: Cursor spotlight | Mouse spotlight | Spotlight hover | Holofote do cursor | Foco de luz no cursor | Luz que segue o mouse

<script lang="ts">
  import { MotionStage } from '@codeatlasdev/ui';

  const plans = [
    { name: 'Starter', price: '$0', unit: '/seat', note: '3 projects, community support' },
    { name: 'Team', price: '$24', unit: '/seat', note: 'Unlimited projects, SSO, audit log' },
    { name: 'Scale', price: '$96', unit: '/seat', note: 'Priority support, 99.99% uptime SLA' },
    { name: 'Enterprise', price: 'Custom', unit: '', note: 'Private cloud, dedicated engineer' }
  ];
  // Autoplay waypoints (x%, y%): one per cycle, the light eases between them.
  const waypoints = [
    [16, 26],
    [74, 22],
    [80, 76],
    [24, 72]
  ];

  // Pointer position in percent; null = follow the autoplay waypoint.
  let pointer = $state<{ x: number; y: number } | null>(null);

  function follow(e: PointerEvent) {
    const r = (e.currentTarget as HTMLElement).getBoundingClientRect();
    pointer = {
      x: ((e.clientX - r.left) / r.width) * 100,
      y: ((e.clientY - r.top) / r.height) * 100
    };
  }
</script>

<MotionStage label="Cursor spotlight: a soft light follows the pointer across a pricing grid" loop={1600}>
  {#snippet children(m)}
    {@const wp = waypoints[m.cycle % waypoints.length]}
    {@const at = pointer ?? { x: wp[0], y: wp[1] }}
    <div
      class="spot"
      class:spot--run={!m.reduced}
      role="presentation"
      style="--x:{m.reduced ? 50 : at.x}%; --y:{m.reduced ? 40 : at.y}%"
      onpointermove={follow}
      onpointerleave={() => (pointer = null)}
    >
      <ul class="spot__grid">
        {#each plans as p}
          <li class="spot__cell">
            <h3 class="spot__name">{p.name}</h3>
            <p class="spot__price">{p.price}<span>{p.unit}</span></p>
            <p class="spot__note">{p.note}</p>
          </li>
        {/each}
      </ul>
      <!-- Two layers share one radial mask: a brand glow, and a grid of lines that only shows inside it. -->
      <div class="spot__glow" aria-hidden="true"></div>
      <div class="spot__lines" aria-hidden="true"></div>
    </div>
  {/snippet}
</MotionStage>
