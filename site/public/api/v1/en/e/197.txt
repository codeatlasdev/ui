Element: #197 Custom zones (charts/gauge)
What it shows: Reverse the default zones when a high value is bad, such as CPU load.
Tags: custom, zones, card, grid, gaugechart
Needs: @codeatlasdev/ui@4.1.0 + echarts
Components: Card, Grid, GaugeChart
Live demo: https://ui.codeatlas.com.br/docs/charts/gauge#custom-zones

<script lang="ts">
  import { Card, Grid } from '@codeatlasdev/ui';
  import { GaugeChart, type GaugeZone } from '@codeatlasdev/ui/charts';

  // `until` is a fraction of min..max; `color` is a semantic name or any CSS color.
  const zones: GaugeZone[] = [
    { until: 0.5, color: 'success' },
    { until: 0.8, color: 'warning' },
    { until: 1, color: 'error' },
  ];
  const metrics = [
    { label: 'CPU', value: 67 },
    { label: 'Memory', value: 82 },
    { label: 'Disk', value: 45 },
  ];
</script>

<Grid class="grid-auto-200" gap="4">
  {#each metrics as m}
    <Card material="glass">
      <p class="label">{m.label}</p>
      <GaugeChart value={m.value} {zones} height="200px" />
    </Card>
  {/each}
</Grid>

<style>
  :global(.grid-auto-200) { grid-template-columns: repeat(auto-fit, minmax(200px, 1fr)); }
  .label {
    text-align: center;
    font-size: var(--ca-font-size-caption);
    color: var(--ca-text-secondary);
    font-weight: 600;
    letter-spacing: 0.05em;
    text-transform: uppercase;
    margin: 0 0 var(--ca-space-1);
  }
</style>
