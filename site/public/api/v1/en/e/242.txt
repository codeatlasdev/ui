Element: #242 Gallery with video, caption and download (feedback/lightbox)
What it shows: A lightbox gallery with three items, two images and one video, each with a caption and some with meta text.
When to use: Browsing attachments or media in a thread, with captions and authors, using next and previous navigation.
Spec:
- `<Lightbox bind:open bind:index {items} />` with `items: LightboxItem[]`
- Three items: image Moodboard, v1, video `type: 'video'` Short clip with a `poster`, image Moodboard, v2
- Fields used: `src`, `alt`, `caption`, `meta`; the third item has no `meta`
- Trigger resets `index = 0; open = true;` and is labelled `Open gallery ({items.length})`
Tags: lightbox, gallery, video, caption, carousel, media
Needs: @codeatlasdev/ui@4.1.0
Components: Button, Lightbox
Live demo: https://ui.codeatlas.com.br/docs/feedback/lightbox#gallery-with-video-caption-and-download

<script lang="ts">
  import { Button, Lightbox, type LightboxItem } from '@codeatlasdev/ui';

  const svg = (fill: string, shape: string) =>
    'data:image/svg+xml;utf8,' +
    encodeURIComponent(
      `<svg xmlns="http://www.w3.org/2000/svg" width="640" height="400"><rect width="640" height="400" fill="${fill}"/>${shape}</svg>`
    );

  const items: LightboxItem[] = [
    {
      src: svg('#334155', '<circle cx="320" cy="200" r="90" fill="#94a3b8"/>'),
      alt: 'Gray circle on a dark background',
      caption: 'Moodboard, v1',
      meta: 'Maria, 10:42',
    },
    {
      type: 'video',
      src: 'https://interactive-examples.mdn.mozilla.net/media/cc0-videos/flower.mp4',
      poster: svg('#0f766e', '<polygon points="280,140 280,260 380,200" fill="#99f6e4"/>'),
      alt: 'Flower blooming',
      caption: 'Short clip',
      meta: 'Bruno, 10:44',
    },
    {
      src: svg('#7c2d12', '<rect x="220" y="120" width="200" height="160" fill="#fdba74"/>'),
      alt: 'Orange square on a brown background',
      caption: 'Moodboard, v2',
    },
  ];

  let open = $state(false);
  let index = $state(0);
</script>

<Button variant="outlined" onclick={() => { index = 0; open = true; }}>Open gallery ({items.length})</Button>
<Lightbox bind:open bind:index {items} />
