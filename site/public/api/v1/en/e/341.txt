Element: #341 Press spring (motion/micro)
What it shows: A button that scales to 0.94 on press and overshoots past 1 on release before settling, driven by a Spring.
When to use: Any tappable surface (pay, send, buy) that needs a physical press feel without a color change.
Spec:
- `MotionStage` from `@codeatlasdev/ui` with `loop={1800}`
- `new Spring(1, { stiffness: 0.35, damping: 0.42 })` is the scale
- `scale.target = 0.94` on pointerdown and on Space or Enter keydown
- `scale.target = 1` on pointerup, pointerleave, pointercancel and keyup: the low damping overshoots
- `transform: scale(var(--s, 1))` is the only animated property
- Under reduced motion `--s` stays at 1
Motion:
- transform: scale: spring spring spring stiffness: 0.35, damping: 0.42
Names: Press spring | Press scale | Efeito de pressionar | Pression élastique
Tags: motion, press, button, spring, tap, feedback
Needs: @codeatlasdev/ui@4.3.0
Components: MotionStage
Live demo: https://ui.codeatlas.com.br/docs/motion/micro#press-spring

<script lang="ts">
  import { MotionStage } from '@codeatlasdev/ui';
  import { Spring } from 'svelte/motion';

  // Press: scale to 0.94 fast (stiff), release overshoots past 1 then settles (low damping).
  const scale = new Spring(1, { stiffness: 0.35, damping: 0.42 });
  let pressed = $state(false);
  let count = $state(0);

  function down() {
    pressed = true;
    scale.target = 0.94;
  }
  function up(user = true) {
    if (!pressed) return;
    pressed = false;
    scale.target = 1;
    if (user) count += 1;
  }

  // Autoplay presses the button for 160ms every cycle.
  function autopress(node: HTMLElement) {
    const t1 = setTimeout(down, 500);
    const t2 = setTimeout(() => up(false), 660);
    // Only the timers go: after a takeover the press belongs to the user.
    return () => {
      clearTimeout(t1);
      clearTimeout(t2);
    };
  }
</script>

<MotionStage label="Press spring: a button scales down on press and overshoots on release" loop={1800}>
  {#snippet children(m)}
    {#key m.cycle}
      <div class="press" {@attach (node) => (m.playing && !m.reduced ? autopress(node) : undefined)}>
        <button
          type="button"
          class="press__btn"
          class:press__btn--down={pressed}
          style="--s:{m.reduced ? 1 : scale.current}"
          onpointerdown={down}
          onpointerup={() => up()}
          onpointerleave={() => up()}
          onpointercancel={() => up()}
          onkeydown={(e) => (e.key === ' ' || e.key === 'Enter') && !e.repeat && down()}
          onkeyup={(e) => (e.key === ' ' || e.key === 'Enter') && up()}
        >
          Pay $48.00
        </button>
        <p class="press__count" aria-live="polite">Pressed {count} {count === 1 ? 'time' : 'times'}</p>
      </div>
    {/key}
  {/snippet}
</MotionStage>
