Element: #381 Grid ripple (motion/hero-ambient)
What it shows: A grid of dots where a click sends a wave outward: each dot swells and fades back after a delay equal to its distance from the click.
When to use: Empty states, hero backdrops and interactive posters where a click or tap should feel like it disturbs a surface.
Spec:
- `MotionStage` with `loop={2200}`; each cycle starts a ripple from the next entry of `origins`
- The delay is the distance in cells times 70: `--d:{Math.round(dist * 70)}ms`, passed to `animation: rip-wave 700ms cubic-bezier(0.2, 0, 0, 1) var(--d) both`
- `rip-wave` keyframes: `transform: scale(2.6)` at the peak, then back to `scale(1)`, colored with `var(--ca-brand-fill)` at the peak
- `{#key src.n}` remounts the dots on every click so all delays restart from the new origin
- Under reduced motion `class:rip__dot--run={!m.reduced}` drops the animation and every dot stays at rest
Motion:
- transform: scale, background-color: 700ms cubic-bezier(0.2, 0, 0, 1) stagger 70ms
Names: Grid ripple | Dot ripple | Click ripple | Ripple from point | Ondulação em grade | Onda de pontos a partir do clique | Ondas en cuadrícula

<script lang="ts">
  import { MotionStage } from '@codeatlasdev/ui';

  const COLS = 14;
  const ROWS = 8;
  const cells = Array.from({ length: COLS * ROWS }, (_, i) => ({ c: i % COLS, r: Math.floor(i / COLS) }));
  // Autoplay origins (col, row): a new ripple starts from the next one every cycle.
  const origins = [
    [3, 2],
    [10, 5],
    [6, 1],
    [11, 2]
  ];

  // The ripple is one number pair; the key restarts the CSS animation on every click.
  let click = $state<{ c: number; r: number; n: number } | null>(null);

  function at(e: PointerEvent) {
    // Measure the dot grid itself, not the padded frame, so the cell under the pointer is exact.
    const grid = (e.currentTarget as HTMLElement).querySelector('.rip__grid');
    if (!grid) return;
    const r = grid.getBoundingClientRect();
    click = {
      c: Math.min(COLS - 1, Math.max(0, Math.floor(((e.clientX - r.left) / r.width) * COLS))),
      r: Math.min(ROWS - 1, Math.max(0, Math.floor(((e.clientY - r.top) / r.height) * ROWS))),
      n: (click?.n ?? 0) + 1
    };
  }
</script>

<MotionStage label="Grid ripple: a wave of dots spreads from the point that was clicked" loop={2200}>
  {#snippet children(m)}
    {@const o = origins[m.cycle % origins.length]}
    {@const src = click ?? { c: o[0], r: o[1], n: m.cycle }}
    <div class="rip" role="presentation" onpointerdown={at}>
      <p class="rip__hint">Click anywhere on the grid</p>
      <!-- Keyed: a new click (or cycle) remounts the dots so every delay restarts from the new origin. -->
      {#key src.n}
        <div class="rip__grid" style="--cols:{COLS}" aria-hidden="true">
          {#each cells as d}
            {@const dist = Math.hypot(d.c - src.c, d.r - src.r)}
            <i class="rip__dot" class:rip__dot--run={!m.reduced} style="--d:{Math.round(dist * 70)}ms"></i>
          {/each}
        </div>
      {/key}
    </div>
  {/snippet}
</MotionStage>
